safe_z <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  if (sum(ok) < 2) {
    out[ok] <- 0
    return(out)
  }
  
  m <- mean(x[ok], na.rm = TRUE)
  s <- stats::sd(x[ok], na.rm = TRUE)
  
  if (!is.finite(s) || s == 0) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - m) / s
  }
  
  out
}

norm_key <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- tolower(x)
  x <- gsub("[[:space:][:punct:]_\\-]+", "", x, perl = TRUE)
  x
}

norm_cell <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- tolower(x)
  x <- gsub("[^a-z0-9]+", "", x, perl = TRUE)
  x
}

make_id <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- toupper(x)
  x <- gsub("\\s+", "", x)
  x <- gsub("^L_", "", x)
  x <- gsub("^R_", "", x)
  x <- gsub("^HNAS_", "", x)
  x <- gsub("^HNAS", "", x)
  x[x %in% c("", "NA", "NAN", "NULL")] <- NA_character_
  x
}

sanitize_filename <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- iconv(x, from = "", to = "ASCII//TRANSLIT", sub = "_")
  x <- gsub("[\r\n\t]+", " ", x)
  x <- gsub("\\s+", "_", x)
  x <- gsub("[/\\\\:*?\"<>|]", "_", x, perl = TRUE)
  x <- gsub("[^A-Za-z0-9_.-]+", "_", x, perl = TRUE)
  x <- gsub("_+", "_", x)
  x <- gsub("^[_\\.]+|[_\\.]+$", "", x)
  ifelse(is.na(x) | x == "", "unknown", x)
}

normalize_group <- function(x) {
  y <- tolower(trimws(as.character(x)))
  y <- gsub("[[:space:]_\\-]+", "", y)
  
  dplyr::case_when(
    y %in% c("hotspot", "hot", "core") ~ "Hotspot",
    y %in% c("near") ~ "Near",
    y %in% c("middle", "mid") ~ "Middle",
    y %in% c("far") ~ "Far",
    TRUE ~ NA_character_
  )
}

normalize_section <- function(x) {
  y <- as.character(x)
  k <- norm_key(y)
  
  dplyr::case_when(
    k %in% c("lrcombined", "lr", "combined") ~ "LR_combined",
    grepl("lrcombined", k) ~ "LR_combined",
    grepl("combined", k) ~ "LR_combined",
    k == "l" ~ "L",
    k == "r" ~ "R",
    grepl("l$", k) & !grepl("lr|combined", k) ~ "L",
    grepl("r$", k) & !grepl("lr|combined", k) ~ "R",
    TRUE ~ y
  )
}

find_first_col <- function(df, candidates) {
  hit <- candidates[candidates %in% colnames(df)]
  if (length(hit) == 0) return(NA_character_)
  hit[1]
}

get_col_or_na <- function(df, col) {
  if (is.na(col) || !col %in% colnames(df)) {
    return(rep(NA_character_, nrow(df)))
  }
  as.character(df[[col]])
}

format_p <- function(p) {
  if (is.na(p) || !is.finite(p)) return("p = NA")
  if (p < 2.2e-16) return("p < 2.2e-16")
  if (p < 0.001) return("p < 0.001")
  paste0("p = ", signif(p, 3))
}

p_stars <- function(p) {
  ifelse(
    is.na(p), "",
    ifelse(
      p < 0.001, "***",
      ifelse(
        p < 0.01, "**",
        ifelse(
          p < 0.05, "*",
          ifelse(p < 0.1, "†", "")
        )
      )
    )
  )
}

module_alias_key <- function(x) {
  k <- norm_key(x)
  out <- k
  
  out[grepl("otherapcaf", k)] <- "nosenescenceapcaf"
  out[grepl("nosenescenceapcaf", k)] <- "nosenescenceapcaf"
  out[grepl("nonsenescenceapcaf", k)] <- "nosenescenceapcaf"
  
  out[grepl("presenescenceapcaf", k)] <- "presenescenceapcaf"
  out[grepl("presenescentapcaf", k)] <- "presenescenceapcaf"
  out[grepl("presenescence", k)] <- "presenescenceapcaf"
  out[grepl("presenescent", k)] <- "presenescenceapcaf"
  
  out[grepl("ecmsenescenceapcaf", k)] <- "ecmsenescenceapcaf"
  out[grepl("ecmsenescentapcaf", k)] <- "ecmsenescenceapcaf"
  
  out[grepl("saspsenescenceapcaf", k)] <- "saspsenescenceapcaf"
  out[grepl("saspsenescentapcaf", k)] <- "saspsenescenceapcaf"
  
  out[k %in% c("senescenceapcaf", "senescentapcaf")] <- "senescenceapcaf"
  
  out
}

fix_module_label <- function(x) {
  k <- norm_key(x)
  
  if (
    grepl("derived", k) |
    grepl("markerassociated", k) |
    grepl("associatedecm", k) |
    grepl("associatedsasp", k)
  ) {
    return(NA_character_)
  }
  
  if (grepl("otherapcaf", k)) return("no-senescence apCAF")
  if (grepl("nosenescence|nonsenescence|nosenescent|nonsenescent", k)) return("no-senescence apCAF")
  
  if (grepl("presenescence|presenescent", k)) return("pre-senescence apCAF")
  
  if (grepl("saspsenescence|saspsenescent", k)) return("SASP senescence apCAF")
  if (grepl("ecmsenescence|ecmsenescent", k)) return("ECM senescence apCAF")
  
  if (k %in% c("senescenceapcaf", "senescentapcaf")) return("senescence apCAF")
  
  if (
    grepl("senescence", k) &
    !grepl("pre|presen|ecm|sasp|no|non|other", k)
  ) {
    return("senescence apCAF")
  }
  
  NA_character_
}

module_priority <- function(raw_label, final_label) {
  k <- norm_key(raw_label)
  
  if (is.na(final_label)) return(999L)
  
  if (final_label == "no-senescence apCAF") {
    if (grepl("otherapcaf", k)) return(1L)
    if (grepl("nosenescence|nonsenescence", k)) return(2L)
    return(9L)
  }
  
  if (final_label == "pre-senescence apCAF" && grepl("pre", k)) return(1L)
  if (final_label == "senescence apCAF" && k %in% c("senescenceapcaf", "senescentapcaf")) return(1L)
  if (final_label == "ECM senescence apCAF" && grepl("ecm", k)) return(1L)
  if (final_label == "SASP senescence apCAF" && grepl("sasp", k)) return(1L)
  
  9L
}

pick_score_col <- function(df) {
  preferred <- c(
    "module_score",
    "marker_score",
    "module_score_z",
    "marker_score_z",
    "score",
    "spot_score",
    "UCell_score",
    "ucell_score"
  )
  
  hit <- preferred[preferred %in% colnames(df)]
  
  if (length(hit) > 0) {
    return(hit[1])
  }
  
  cn <- colnames(df)
  
  cand <- cn[
    grepl("module.*score|marker.*score|ucell.*score|score", cn, ignore.case = TRUE) &
      !grepl("xcell|distance|near|far|hotspot|group|mean|median|pct|count|n_", cn, ignore.case = TRUE)
  ]
  
  if (length(cand) > 0) {
    return(cand[1])
  }
  
  NA_character_
}

get_module_label_from_spot <- function(spot_file) {
  dfh <- tryCatch(
    read.csv(spot_file, stringsAsFactors = FALSE, check.names = FALSE, nrows = 30),
    error = function(e) data.frame()
  )
  
  if (nrow(dfh) > 0) {
    module_col <- find_first_col(dfh, c("object_label", "module_label", "object_id", "module"))
    
    if (!is.na(module_col)) {
      val <- unique(na.omit(as.character(dfh[[module_col]])))
      val <- val[val != ""]
      if (length(val) > 0) return(val[1])
    }
  }
  
  basename(dirname(dirname(spot_file)))
}

find_xcell_long_for_spot <- function(spot_file, raw_label, all_long_files) {
  same_dir <- dirname(spot_file)
  
  cand_same <- list.files(
    same_dir,
    pattern = "xCell_score_long.*\\.csv$|_xCell_score_long\\.csv$",
    full.names = TRUE,
    ignore.case = TRUE
  )
  
  if (length(cand_same) > 0) {
    return(cand_same[1])
  }
  
  alias <- module_alias_key(raw_label)
  raw_norm <- norm_key(raw_label)
  path_norm <- norm_key(all_long_files)
  
  hit <- all_long_files[
    grepl(alias, path_norm, fixed = TRUE) |
      grepl(raw_norm, path_norm, fixed = TRUE)
  ]
  
  if (length(hit) > 0) return(hit[1])
  
  NA_character_
}

read_module_score_by_distance <- function(spot_file, module_label_final) {
  
  message("读取模块 spot 文件：", spot_file)
  
  df <- read.csv(
    spot_file,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  section_col <- find_first_col(df, c("section", "sample_short"))
  group_col <- find_first_col(df, c("xcell_group", "distance_group", "group", "group_label"))
  score_col <- pick_score_col(df)
  circle_col <- find_first_col(df, c("pseudo_circle_id", "circle_id", "circle"))
  hnas_col <- find_first_col(df, c("HNas_code", "hnas_code", "HNas"))
  spot_col <- find_first_col(df, c("spot_id", "barcode"))
  
  if (is.na(section_col)) stop("spot 文件缺少 section/sample_short 列：", spot_file)
  if (is.na(group_col)) stop("spot 文件缺少 xcell_group/distance_group/group 列：", spot_file)
  if (is.na(score_col)) stop("spot 文件缺少 module score 列：", spot_file)
  if (is.na(circle_col) && is.na(hnas_col)) stop("spot 文件缺少 pseudo_circle_id/circle_id/HNas_code：", spot_file)
  
  out0 <- df %>%
    dplyr::mutate(
      module_label = module_label_final,
      section = normalize_section(.data[[section_col]]),
      xcell_group = normalize_group(.data[[group_col]]),
      module_score = as.numeric(.data[[score_col]]),
      circle_raw = get_col_or_na(df, circle_col),
      hnas_raw = get_col_or_na(df, hnas_col),
      circle_clean = make_id(circle_raw),
      HNas_code_clean = make_id(hnas_raw),
      unit_id = dplyr::case_when(
        !is.na(HNas_code_clean) & HNas_code_clean != "" ~ HNas_code_clean,
        !is.na(circle_clean) & circle_clean != "" ~ circle_clean,
        TRUE ~ NA_character_
      ),
      spot_id = if (!is.na(spot_col)) as.character(.data[[spot_col]]) else as.character(dplyr::row_number())
    ) %>%
    dplyr::filter(
      section %in% c("L", "R", "LR_combined"),
      xcell_group %in% GROUP_ORDER,
      !is.na(unit_id),
      unit_id != "",
      is.finite(module_score)
    )
  
  if (nrow(out0) == 0) {
    stop("模块 spot 表标准化后为空：", spot_file)
  }
  
  ## section-specific score
  sec_score <- out0 %>%
    dplyr::group_by(
      module_label,
      section,
      unit_id,
      xcell_group
    ) %>%
    dplyr::summarise(
      module_score = mean(module_score, na.rm = TRUE),
      module_score_median = median(module_score, na.rm = TRUE),
      n_spots_module = dplyr::n_distinct(spot_id),
      circle_id_examples = paste(unique(na.omit(circle_clean))[1:min(5, length(unique(na.omit(circle_clean))))], collapse = ";"),
      HNas_examples = paste(unique(na.omit(HNas_code_clean))[1:min(5, length(unique(na.omit(HNas_code_clean))))], collapse = ";"),
      score_col_used = score_col,
      spot_group_file = spot_file,
      .groups = "drop"
    )
  
  ## LR_combined score：如果原文件没有 LR_combined，则按 unit_id + distance group 聚合 L/R
  lr_score <- out0 %>%
    dplyr::group_by(
      module_label,
      unit_id,
      xcell_group
    ) %>%
    dplyr::summarise(
      section = "LR_combined",
      module_score = mean(module_score, na.rm = TRUE),
      module_score_median = median(module_score, na.rm = TRUE),
      n_spots_module = dplyr::n_distinct(spot_id),
      circle_id_examples = paste(unique(na.omit(circle_clean))[1:min(5, length(unique(na.omit(circle_clean))))], collapse = ";"),
      HNas_examples = paste(unique(na.omit(HNas_code_clean))[1:min(5, length(unique(na.omit(HNas_code_clean))))], collapse = ";"),
      score_col_used = score_col,
      spot_group_file = spot_file,
      .groups = "drop"
    ) %>%
    dplyr::select(
      module_label,
      section,
      unit_id,
      xcell_group,
      module_score,
      module_score_median,
      n_spots_module,
      circle_id_examples,
      HNas_examples,
      score_col_used,
      spot_group_file
    )
  
  dplyr::bind_rows(sec_score, lr_score) %>%
    dplyr::mutate(
      xcell_group = factor(xcell_group, levels = GROUP_ORDER)
    ) %>%
    dplyr::arrange(module_label, section, unit_id, xcell_group) %>%
    dplyr::mutate(
      xcell_group = as.character(xcell_group)
    )
}

## 9. 读取 xCell long，压缩成 immune_score / stromal_score
############################################################

read_xcell_immune_stromal <- function(xcell_long_file, module_label_final) {
  
  message("读取 xCell long 文件：", xcell_long_file)
  
  df <- read.csv(
    xcell_long_file,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  cell_col <- find_first_col(df, c("xcell_celltype", "cell_type", "celltype"))
  score_col <- find_first_col(df, c("xcell_score", "score", "mean_score"))
  sample_col <- find_first_col(df, c("pseudo_sample_id", "sample_id", "pseudosample_id"))
  section_col <- find_first_col(df, c("section", "sample_short"))
  group_col <- find_first_col(df, c("xcell_group", "distance_group", "group", "group_label"))
  circle_col <- find_first_col(df, c("pseudo_circle_id", "circle_id", "circle"))
  hnas_col <- find_first_col(df, c("HNas_code", "hnas_code", "HNas"))
  nspot_col <- find_first_col(df, c("n_spots", "n_spot", "spot_count"))
  
  if (is.na(cell_col)) stop("xCell long 缺少 xcell_celltype/cell_type 列：", xcell_long_file)
  if (is.na(score_col)) stop("xCell long 缺少 xcell_score/score 列：", xcell_long_file)
  if (is.na(group_col)) stop("xCell long 缺少 xcell_group/distance_group/group 列：", xcell_long_file)
  
  x0 <- df %>%
    dplyr::mutate(
      module_label = module_label_final,
      pseudo_sample_id = if (!is.na(sample_col)) as.character(.data[[sample_col]]) else paste0("sample_", dplyr::row_number()),
      section = if (!is.na(section_col)) normalize_section(.data[[section_col]]) else "LR_combined",
      xcell_group = normalize_group(.data[[group_col]]),
      xcell_celltype = as.character(.data[[cell_col]]),
      xcell_celltype_norm = norm_cell(xcell_celltype),
      xcell_score = as.numeric(.data[[score_col]]),
      circle_raw = get_col_or_na(df, circle_col),
      hnas_raw = get_col_or_na(df, hnas_col),
      circle_clean = make_id(circle_raw),
      HNas_code_clean = make_id(hnas_raw),
      unit_id = dplyr::case_when(
        !is.na(HNas_code_clean) & HNas_code_clean != "" ~ HNas_code_clean,
        !is.na(circle_clean) & circle_clean != "" ~ circle_clean,
        TRUE ~ make_id(pseudo_sample_id)
      ),
      n_spots_xcell = if (!is.na(nspot_col)) as.numeric(.data[[nspot_col]]) else NA_real_
    ) %>%
    dplyr::filter(
      section %in% c("L", "R", "LR_combined"),
      xcell_group %in% GROUP_ORDER,
      is.finite(xcell_score),
      !is.na(unit_id),
      unit_id != ""
    )
  
  if (nrow(x0) == 0) {
    stop("xCell long 标准化后为空：", xcell_long_file)
  }
  
  ## 64 细胞类型归并
  x64 <- x0 %>%
    dplyr::left_join(
      celltype_to_category %>%
        dplyr::select(xcell_celltype_norm, cell_category),
      by = "xcell_celltype_norm"
    ) %>%
    dplyr::filter(
      cell_category %in% c("Immune cells", "Stromal cells")
    )
  
  if (nrow(x64) > 0) {
    
    x64_z <- x64 %>%
      dplyr::group_by(xcell_celltype_norm) %>%
      dplyr::mutate(
        xcell_score_z = safe_z(xcell_score)
      ) %>%
      dplyr::ungroup()
    
    cat_long <- x64_z %>%
      dplyr::group_by(
        module_label,
        section,
        unit_id,
        xcell_group,
        cell_category
      ) %>%
      dplyr::summarise(
        category_score_raw = mean(xcell_score, na.rm = TRUE),
        category_score_z = mean(xcell_score_z, na.rm = TRUE),
        n_celltypes_used = dplyr::n_distinct(xcell_celltype_norm),
        xcell_source = "manual_64cell_category",
        xcell_long_file = xcell_long_file,
        .groups = "drop"
      )
    
  } else {
    
    if (!ALLOW_AGGREGATE_FALLBACK) {
      stop("没有匹配到 64 类 immune/stromal xCell 细胞类型：", xcell_long_file)
    }
    
    message("没有匹配到 64 类细胞类型，使用 ImmuneScore/StromaScore 兜底：", xcell_long_file)
    
    xagg <- x0 %>%
      dplyr::filter(
        xcell_celltype %in% c("ImmuneScore", "StromaScore")
      ) %>%
      dplyr::mutate(
        cell_category = dplyr::case_when(
          xcell_celltype == "ImmuneScore" ~ "Immune cells",
          xcell_celltype == "StromaScore" ~ "Stromal cells",
          TRUE ~ NA_character_
        )
      ) %>%
      dplyr::filter(
        cell_category %in% c("Immune cells", "Stromal cells")
      )
    
    if (nrow(xagg) == 0) {
      stop("既没有 64 类细胞类型，也没有 ImmuneScore/StromaScore：", xcell_long_file)
    }
    
    xagg_z <- xagg %>%
      dplyr::group_by(xcell_celltype_norm) %>%
      dplyr::mutate(
        xcell_score_z = safe_z(xcell_score)
      ) %>%
      dplyr::ungroup()
    
    cat_long <- xagg_z %>%
      dplyr::group_by(
        module_label,
        section,
        unit_id,
        xcell_group,
        cell_category
      ) %>%
      dplyr::summarise(
        category_score_raw = mean(xcell_score, na.rm = TRUE),
        category_score_z = mean(xcell_score_z, na.rm = TRUE),
        n_celltypes_used = dplyr::n_distinct(xcell_celltype_norm),
        xcell_source = "aggregate_ImmuneScore_StromaScore_fallback",
        xcell_long_file = xcell_long_file,
        .groups = "drop"
      )
  }
  
  cat_wide_z <- cat_long %>%
    dplyr::select(
      module_label,
      section,
      unit_id,
      xcell_group,
      cell_category,
      category_score_z
    ) %>%
    tidyr::pivot_wider(
      names_from = cell_category,
      values_from = category_score_z
    )
  
  cat_wide_raw <- cat_long %>%
    dplyr::select(
      module_label,
      section,
      unit_id,
      xcell_group,
      cell_category,
      category_score_raw
    ) %>%
    tidyr::pivot_wider(
      names_from = cell_category,
      values_from = category_score_raw
    )
  
  meta <- cat_long %>%
    dplyr::group_by(
      module_label,
      section,
      unit_id,
      xcell_group
    ) %>%
    dplyr::summarise(
      n_immune_celltypes_used = sum(n_celltypes_used[cell_category == "Immune cells"], na.rm = TRUE),
      n_stromal_celltypes_used = sum(n_celltypes_used[cell_category == "Stromal cells"], na.rm = TRUE),
      xcell_source = paste(unique(xcell_source), collapse = ";"),
      xcell_long_file = paste(unique(xcell_long_file), collapse = ";"),
      .groups = "drop"
    )
  
  out <- cat_wide_z %>%
    dplyr::rename(
      immune_score_zmean = `Immune cells`,
      stromal_score_zmean = `Stromal cells`
    ) %>%
    dplyr::left_join(
      cat_wide_raw %>%
        dplyr::rename(
          immune_score_rawmean = `Immune cells`,
          stromal_score_rawmean = `Stromal cells`
        ),
      by = c("module_label", "section", "unit_id", "xcell_group")
    ) %>%
    dplyr::left_join(
      meta,
      by = c("module_label", "section", "unit_id", "xcell_group")
    ) %>%
    dplyr::mutate(
      immune_score = if (XCELL_SCORE_MODE == "raw_mean") immune_score_rawmean else immune_score_zmean,
      stromal_score = if (XCELL_SCORE_MODE == "raw_mean") stromal_score_rawmean else stromal_score_zmean
    ) %>%
    dplyr::filter(
      is.finite(immune_score),
      is.finite(stromal_score)
    )
  
  ## 如果原始 xCell long 没有 LR_combined，则额外生成一个 pooled LR_combined
  if (!any(out$section == "LR_combined")) {
    pooled <- out %>%
      dplyr::group_by(
        module_label,
        unit_id,
        xcell_group
      ) %>%
      dplyr::summarise(
        section = "LR_combined",
        immune_score_zmean = mean(immune_score_zmean, na.rm = TRUE),
        stromal_score_zmean = mean(stromal_score_zmean, na.rm = TRUE),
        immune_score_rawmean = mean(immune_score_rawmean, na.rm = TRUE),
        stromal_score_rawmean = mean(stromal_score_rawmean, na.rm = TRUE),
        immune_score = mean(immune_score, na.rm = TRUE),
        stromal_score = mean(stromal_score, na.rm = TRUE),
        n_immune_celltypes_used = max(n_immune_celltypes_used, na.rm = TRUE),
        n_stromal_celltypes_used = max(n_stromal_celltypes_used, na.rm = TRUE),
        xcell_source = paste(unique(xcell_source), collapse = ";"),
        xcell_long_file = paste(unique(xcell_long_file), collapse = ";"),
        .groups = "drop"
      ) %>%
      dplyr::select(
        module_label,
        section,
        unit_id,
        xcell_group,
        immune_score_zmean,
        stromal_score_zmean,
        immune_score_rawmean,
        stromal_score_rawmean,
        n_immune_celltypes_used,
        n_stromal_celltypes_used,
        xcell_source,
        xcell_long_file,
        immune_score,
        stromal_score
      )
    
    out <- dplyr::bind_rows(out, pooled)
  }
  
  out %>%
    dplyr::mutate(
      xcell_group = factor(xcell_group, levels = GROUP_ORDER)
    ) %>%
    dplyr::arrange(module_label, section, unit_id, xcell_group) %>%
    dplyr::mutate(
      xcell_group = as.character(xcell_group)
    )
}

calc_cor <- function(df, x_col, y_col = "module_score", method = "pearson") {
  
  x <- as.numeric(df[[x_col]])
  y <- as.numeric(df[[y_col]])
  
  ok <- is.finite(x) & is.finite(y)
  n_ok <- sum(ok)
  
  if (n_ok < MIN_N_COR) {
    return(data.frame(
      n = n_ok,
      estimate = NA_real_,
      p_value = NA_real_,
      conf_low = NA_real_,
      conf_high = NA_real_,
      skipped = TRUE,
      skip_reason = paste0("n < ", MIN_N_COR),
      stringsAsFactors = FALSE
    ))
  }
  
  if (length(unique(x[ok])) < 3 || length(unique(y[ok])) < 3) {
    return(data.frame(
      n = n_ok,
      estimate = NA_real_,
      p_value = NA_real_,
      conf_low = NA_real_,
      conf_high = NA_real_,
      skipped = TRUE,
      skip_reason = "too_few_unique_values",
      stringsAsFactors = FALSE
    ))
  }
  
  ct <- suppressWarnings(
    stats::cor.test(
      x[ok],
      y[ok],
      method = method,
      exact = FALSE
    )
  )
  
  conf_low <- NA_real_
  conf_high <- NA_real_
  
  if (!is.null(ct$conf.int)) {
    conf_low <- as.numeric(ct$conf.int[1])
    conf_high <- as.numeric(ct$conf.int[2])
  }
  
  data.frame(
    n = n_ok,
    estimate = as.numeric(ct$estimate),
    p_value = as.numeric(ct$p.value),
    conf_low = conf_low,
    conf_high = conf_high,
    skipped = FALSE,
    skip_reason = NA_character_,
    stringsAsFactors = FALSE
  )
}

        df_sub <- cor_input %>%
          dplyr::filter(
            section == sec,
            module_label == mod,
            xcell_group == grp
          )
        
        out_list <- list()
        
        for (pred_col in c("immune_score", "stromal_score")) {
          
          pred_label <- dplyr::case_when(
            pred_col == "immune_score" ~ "Immune score",
            pred_col == "stromal_score" ~ "Stromal score",
            TRUE ~ pred_col
          )
          
          pearson_i <- calc_cor(
            df = df_sub,
            x_col = pred_col,
            y_col = "module_score",
            method = "pearson"
          )
          
          pearson_i$section <- sec
          pearson_i$module_label <- mod
          pearson_i$xcell_group <- grp
          pearson_i$predictor <- pred_label
          pearson_i$predictor_col <- pred_col

          spearman_i <- calc_cor(

          out_list[[length(out_list) + 1]] <- pearson_i

trend_df <- cor_res %>%
  dplyr::filter(
    method == MAIN_METHOD,
    skipped == FALSE,
    is.finite(estimate)
  ) %>%
  dplyr::group_by(section, module_label, predictor) %>%
  dplyr::summarise(
    n_distance_available = dplyr::n(),
    r_hotspot = estimate[xcell_group == "Hotspot"][1],
    r_near = estimate[xcell_group == "Near"][1],
    r_middle = estimate[xcell_group == "Middle"][1],
    r_far = estimate[xcell_group == "Far"][1],
    r_change_far_minus_hotspot = r_far - r_hotspot,
    trend_slope = ifelse(
      dplyr::n() >= 3,
      coef(lm(estimate ~ distance_order))[2],
      NA_real_
    ),
    trend_direction = dplyr::case_when(
      is.na(trend_slope) ~ NA_character_,
      trend_slope > 0 ~ "correlation increases with distance",
      trend_slope < 0 ~ "correlation decreases with distance",
      TRUE ~ "flat"
    ),
    .groups = "drop"
  ) %>%
  dplyr::mutate(
    module_label = factor(module_label, levels = FINAL_MODULE_ORDER),
    predictor = factor(predictor, levels = PREDICTOR_ORDER)
  ) %>%
  dplyr::arrange(section, module_label, predictor) %>%
  dplyr::mutate(
    module_label = as.character(module_label),
    predictor = as.character(predictor)
  )
