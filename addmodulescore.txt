get_data_mat <- function(obj, assay_use) {
  mat <- tryCatch({
    GetAssayData(obj, assay = assay_use, layer = "data")
  }, error = function(e) {
    GetAssayData(obj, assay = assay_use, slot = "data")
  })
  
  return(mat)
}

data_mat <- tryCatch({
  get_data_mat(caf_obj, ASSAY_USE)
}, error = function(e) {
  NULL
})

if (is.null(data_mat) || nrow(data_mat) == 0 || ncol(data_mat) == 0) {
  cat("\n没有检测到 normalized data，开始 NormalizeData...\n")
  caf_obj <- NormalizeData(
    caf_obj,
    assay = ASSAY_USE,
    normalization.method = "LogNormalize",
    scale.factor = 10000,
    verbose = FALSE
  )
  
  data_mat <- get_data_mat(caf_obj, ASSAY_USE)

############################################################
## 13. 组间统计检验
## 三组整体：Kruskal-Wallis
## 两两比较：Wilcoxon rank-sum test
############################################################

df_score <- caf_obj@meta.data %>%
  as.data.frame() %>%
  dplyr::filter(!is.na(CAF_subtype_for_senescence)) %>%
  dplyr::filter(!is.na(Senescence_score_0_1))

df_score$CAF_subtype_for_senescence <- factor(
  df_score$CAF_subtype_for_senescence,
  levels = c("apCAF", "iCAF", "myCAF")
)

kw_res <- kruskal.test(
  Senescence_score_0_1 ~ CAF_subtype_for_senescence,
  data = df_score
)

pair_list <- list(
  c("apCAF", "iCAF"),
  c("apCAF", "myCAF"),
  c("iCAF", "myCAF")
)

pair_list <- pair_list[
  sapply(pair_list, function(x) all(x %in% df_score$CAF_subtype_for_senescence))
]

pair_res_df <- data.frame()

for (i in seq_along(pair_list)) {
  g1 <- pair_list[[i]][1]
  g2 <- pair_list[[i]][2]
  
  x1 <- df_score$Senescence_score_0_1[
    df_score$CAF_subtype_for_senescence == g1
  ]
  
  x2 <- df_score$Senescence_score_0_1[
    df_score$CAF_subtype_for_senescence == g2
  ]
  
  wt <- wilcox.test(x1, x2, exact = FALSE)
  
  pair_res_df <- rbind(
    pair_res_df,
    data.frame(
      group1 = g1,
      group2 = g2,
      p_value = wt$p.value
    )
  )
}

pair_res_df$p_adj_BH <- p.adjust(pair_res_df$p_value, method = "BH")

format_p <- function(p) {
  if (is.na(p)) {
    return("NA")
  } else if (p < 0.001) {
    return("p < 0.001")
  } else {
    return(paste0("p = ", signif(p, 3)))
  }
}

pair_res_df$p_label <- sapply(pair_res_df$p_adj_BH, format_p)

write.csv(
  pair_res_df,
  file = file.path(DIR_OUT, "06_pairwise_Wilcoxon_rank_sum_test.csv"),
  row.names = FALSE
)

sink(file.path(DIR_OUT, "06_statistics_test_summary.txt"))
cat("Kruskal-Wallis test for three CAF subtypes\n")
print(kw_res)
cat("\n\nPairwise Wilcoxon rank-sum test, BH adjusted\n")
print(pair_res_df)
sink()

cat("\nKruskal-Wallis P value:", kw_res$p.value, "\n")
cat("\n两两 Wilcoxon 秩和检验：\n")
print(pair_res_df)
