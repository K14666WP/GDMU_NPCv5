safe_z <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  m <- mean(x[ok], na.rm = TRUE)
  s <- sd(x[ok], na.rm = TRUE)
  
  if (!is.finite(s) || s == 0) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - m) / s
  }
  
  out
}

minmax_01 <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  mn <- min(x[ok], na.rm = TRUE)
  mx <- max(x[ok], na.rm = TRUE)
  
  if (!is.finite(mx - mn) || mx == mn) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - mn) / (mx - mn)
  }
  
  out
}

clean_gene_key <- function(x) {
  x <- toupper(as.character(x))
  x <- trimws(x)
  x <- gsub("\\s+", "", x)
  x <- gsub("[^A-Z0-9\\-]", "", x)
  x
}

sanitize_name <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("[\r\n\t]+", " ", x)
  x <- gsub("\\s+", "_", x)
  x <- gsub("[^A-Za-z0-9_\\u4e00-\\u9fa5]", "_", x)
  x <- gsub("_+", "_", x)
  x <- gsub("^_|_$", "", x)
  ifelse(x == "", "module", x)
}

dedup_genes_keep_high_mean <- function(mat) {
  gene <- rownames(mat)
  
  if (!anyDuplicated(gene)) {
    return(mat)
  }
  
  gene_mean <- Matrix::rowMeans(mat)
  
  df <- data.frame(
    gene = gene,
    idx = seq_along(gene),
    mean_expr = as.numeric(gene_mean),
    stringsAsFactors = FALSE
  ) %>%
    arrange(gene, desc(mean_expr)) %>%
    distinct(gene, .keep_all = TRUE)
  
  mat2 <- mat[df$idx, , drop = FALSE]
  rownames(mat2) <- df$gene
  
  mat2
}

add_edge_layer <- function(p, edge_df) {
  if (is.null(edge_df) || !is.data.frame(edge_df) || nrow(edge_df) == 0) {
    return(p)
  }
  
  if (all(c("x", "y", "xend", "yend") %in% colnames(edge_df))) {
    p <- p +
      geom_segment(
        data = edge_df,
        aes(x = x, y = y, xend = xend, yend = yend),
        inherit.aes = FALSE,
        color = "black",
        linewidth = EDGE_SIZE,
        alpha = EDGE_ALPHA
      )
  }
  
  p
}

read_marker_modules_from_excel <- function(file) {
  sheets <- readxl::excel_sheets(file)
  
  all_modules <- list()
  all_meta <- data.frame()
  
  for (sh in sheets) {
    raw <- tryCatch(
      as.data.frame(readxl::read_excel(file, sheet = sh, col_names = FALSE)),
      error = function(e) NULL
    )
    
    if (is.null(raw) || nrow(raw) == 0 || ncol(raw) == 0) next
    
    raw <- raw[, colSums(!is.na(raw)) > 0, drop = FALSE]
    
    if (nrow(raw) == 0 || ncol(raw) == 0) next
    
    first_row <- as.character(unlist(raw[1, ]))
    first_row[is.na(first_row)] <- ""
    header_cols <- which(trimws(first_row) != "")
    
    if (length(header_cols) >= 2) {
      for (i in header_cols) {
        label <- trimws(as.character(raw[1, i]))
        module_id <- sanitize_name(label)
        
        genes <- raw[-1, i, drop = TRUE]
        genes <- as.character(genes)
        genes <- trimws(genes)
        genes <- genes[!is.na(genes) & genes != ""]
        genes <- unique(genes)
        
        if (length(genes) == 0) next
        
        if (module_id %in% names(all_modules)) {
          module_id <- paste0(module_id, "_", sanitize_name(sh))
        }
        
        all_modules[[module_id]] <- genes
        
        all_meta <- bind_rows(
          all_meta,
          data.frame(
            sheet = sh,
            module_id = module_id,
            module_label = label,
            input_gene = genes,
            stringsAsFactors = FALSE
          )
        )
      }
    }
    
    if (length(header_cols) < 2) {
      genes <- as.character(unlist(raw))
      genes <- trimws(genes)
      genes <- genes[!is.na(genes) & genes != ""]
      genes <- genes[!genes %in% c("gene", "Gene", "GENE", "marker", "Marker", "MARKER")]
      genes <- unique(genes)
      
      if (length(genes) > 0) {
        module_id <- sanitize_name(sh)
        label <- sh
        
        if (module_id %in% names(all_modules)) {
          module_id <- paste0(module_id, "_sheet")
        }
        
        all_modules[[module_id]] <- genes
        
        all_meta <- bind_rows(
          all_meta,
          data.frame(
            sheet = sh,
            module_id = module_id,
            module_label = label,
            input_gene = genes,
            stringsAsFactors = FALSE
          )
        )
      }
    }
  }
  
  if (length(all_modules) == 0) {
    stop("marker Excel 没有读到任何模块。请检查文件格式。")
  }
  
  list(
    modules = all_modules,
    module_meta = all_meta
  )
}

plot_ucell_module <- function(mid) {

cor_df <- lapply(names(module_list_ucell), function(mid) {
  z_col <- paste0(mid, "__UCell_z")
  raw_col <- paste0(mid, "__UCell")
  
  x <- plot_df$Pseudotime_0_1
  y <- plot_df[[z_col]]
  
  ok <- is.finite(x) & is.finite(y)
  
  if (sum(ok) < 10) {
    return(
      data.frame(
        module_id = mid,
        module_label = module_labels[[mid]],
        n_spots = sum(ok),
        spearman_rho = NA_real_,
        spearman_p = NA_real_,
        stringsAsFactors = FALSE
      )
    )
  }
  
  sp <- suppressWarnings(cor.test(x[ok], y[ok], method = "spearman", exact = FALSE))
  
  data.frame(
    module_id = mid,
    module_label = module_labels[[mid]],
    n_spots = sum(ok),
    spearman_rho = as.numeric(sp$estimate),
    spearman_p = sp$p.value,
    stringsAsFactors = FALSE
  )
}) %>%
  bind_rows() %>%
  mutate(
    spearman_FDR = p.adjust(spearman_p, method = "BH"),
    trend_direction = case_when(
      is.na(spearman_rho) ~ "NA",
      spearman_rho > 0 ~ "increase_along_pseudotime",
      spearman_rho < 0 ~ "decrease_along_pseudotime",
      TRUE ~ "flat"
    )
  ) %>%
  arrange(desc(abs(spearman_rho)))

# z-score标准化
safe_z <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  if (sum(ok) == 0) return(out)
  m <- mean(x[ok], na.rm = TRUE)
  s <- sd(x[ok], na.rm = TRUE)
  if (!is.finite(s) || s == 0) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - m) / s
  }
  out
}

# 0-1归一化
minmax_01 <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  if (sum(ok) == 0) return(out)
  mn <- min(x[ok], na.rm = TRUE)
  mx <- max(x[ok], na.rm = TRUE)
  if (mx == mn) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - mn) / (mx - mn)
  }
  out
}

# 基因名清洗大写去符号
clean_gene_key <- function(x) {
  x <- toupper(trimws(as.character(x)))
  x <- gsub("\\s+", "", x)
  x <- gsub("[^A-Z0-9\\-]", "", x)
  x
}

# 添加轨迹连线edge
add_edge_layer <- function(p, edge_df) {
  if (is.null(edge_df) || !is.data.frame(edge_df) || nrow(edge_df) == 0) return(p)
  if (all(c("x", "y", "xend", "yend") %in% colnames(edge_df))) {
    p <- p + geom_segment(
      data = edge_df,
      aes(x = x, y = y, xend = xend, yend = yend),
      inherit.aes = FALSE,
      color = "black",
      linewidth = EDGE_SIZE,
      alpha = EDGE_ALPHA
    )
  }
  p
}

  group_by(pseudotime_bin) %>%
  summarise(
    n_spots = n(),
    pt_mean = mean(Pseudotime_0_1, na.rm = TRUE),
    z_mean = mean(UCell_z, na.rm = TRUE),
    z_sd = sd(UCell_z, na.rm = TRUE),
    z_se = z_sd / sqrt(n_spots),
    .groups = "drop"
  )

# Spearman相关性
valid_corr <- is.finite(trend_long$Pseudotime_0_1) & is.finite(trend_long$UCell_z)
cor_res <- cor.test(
  x = trend_long$Pseudotime_0_1[valid_corr],
  y = trend_long$UCell_z[valid_corr],
  method = "spearman",
  exact = FALSE
)

# 绘图
p_trend <- ggplot() +
  geom_point(
    data = trend_long,
    aes(Pseudotime_0_1, UCell_z),
    size = 0.25,
    alpha = 0.10,
    color = "grey45"
  ) +
  geom_ribbon(
    data = bin_summary,
    aes(x = pt_mean, ymin = z_mean - z_se, ymax = z_mean + z_se),
    alpha = 0.20,
    fill = MODULE_HIGH
  ) +
  geom_line(
    data = bin_summary,
    aes(pt_mean, z_mean),
    linewidth = 1.0,
    color = MODULE_HIGH
  ) +
  geom_hline(yintercept = 0, linetype = "dashed", linewidth = 0.35, color = "grey40") +
  theme_trend() +
  labs(
    title = TARGET_LABEL,
    subtitle = paste0(
      "Spearman rho = ", round(cor_res$estimate, 3),
      "; P-value = ", signif(cor_res$p.value, 3)
    ),
    x = "Pseudotime 0-1",
    y = "UCell z-score"
  )

# 保存趋势图
save_pdf_png(
  p_trend,
  paste0("UCell_Trend_", TARGET_MODULE_ID),
  width = 6.4,
  height = 4.8,
  outdir = DIR_TREND
)

# ====================== 11. 合并拼图（轨迹图+趋势图） ======================
p_composite <- p_trajectory + p_trend +
  plot_annotation(
    title = paste0(TARGET_LABEL, " UCell score trajectory & pseudotime trend"),
    theme = theme(plot.title = element_text(face = "bold", size = 16))
  )

save_pdf_png(
