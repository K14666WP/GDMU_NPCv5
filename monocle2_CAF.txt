safe_z <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  m <- mean(x[ok], na.rm = TRUE)
  s <- sd(x[ok], na.rm = TRUE)
  
  if (!is.finite(s) || s == 0) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - m) / s
  }
  
  out
}

minmax_01 <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  mn <- min(x[ok], na.rm = TRUE)
  mx <- max(x[ok], na.rm = TRUE)
  
  if (!is.finite(mx - mn) || mx == mn) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - mn) / (mx - mn)
  }
  
  out
}

clean_gene_key <- function(x) {
  x <- toupper(as.character(x))
  x <- trimws(x)
  x <- gsub("\\s+", "", x)
  x <- gsub("[^A-Z0-9\\-]", "", x)
  x
}

sanitize_name <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("[\r\n\t]+", " ", x)
  x <- gsub("\\s+", "_", x)
  x <- gsub("[^A-Za-z0-9_\\u4e00-\\u9fa5]", "_", x)
  x <- gsub("_+", "_", x)
  x <- gsub("^_|_$", "", x)
  ifelse(x == "", "module", x)
}

norm_label <- function(x) {
  x <- tolower(as.character(x))
  x <- gsub("[^a-z0-9]+", "", x)
  x
}

make_key <- function(x) {
  x <- as.character(x)
  x <- basename(x)
  x <- trimws(x)
  x <- gsub("\\s+", "", x)
  x <- gsub("\"", "", x)
  x <- gsub("'", "", x)
  toupper(x)
}

barcode_core <- function(x) {
  x <- make_key(x)
  x <- gsub("^HNASN091SU01_L_", "", x)
  x <- gsub("^HNASN091SU01_R_", "", x)
  x <- gsub("^L_", "", x)
  x <- gsub("^R_", "", x)
  x <- gsub("-[0-9]+$", "", x)
  x
}

make_spot_id <- function(barcode, sample_id) {
  paste0(sample_id, "_", as.character(barcode))
}

dedup_genes_keep_high_mean <- function(mat) {
  gene <- rownames(mat)
  
  if (!anyDuplicated(gene)) return(mat)
  
  gene_mean <- Matrix::rowMeans(mat)
  
  df <- data.frame(
    gene = gene,
    idx = seq_along(gene),
    mean_expr = as.numeric(gene_mean),
    stringsAsFactors = FALSE
  ) %>%
    arrange(gene, desc(mean_expr)) %>%
    distinct(gene, .keep_all = TRUE)
  
  mat2 <- mat[df$idx, , drop = FALSE]
  rownames(mat2) <- df$gene
  
  mat2
}

merge_sparse_by_gene <- function(mat_list) {
  all_genes <- unique(unlist(lapply(mat_list, rownames)))
  all_cells <- unlist(lapply(mat_list, colnames))
  
  out_list <- lapply(mat_list, function(m) {
    miss <- setdiff(all_genes, rownames(m))
    
    if (length(miss) > 0) {
      zero <- Matrix::Matrix(
        0,
        nrow = length(miss),
        ncol = ncol(m),
        sparse = TRUE,
        dimnames = list(miss, colnames(m))
      )
      
      m <- rbind(m, zero)
    }
    
    m[all_genes, , drop = FALSE]
  })
  
  merged <- do.call(cbind, out_list)
  merged[, all_cells, drop = FALSE]
}

read_counts_from_outs <- function(outs_dir, sample_id) {
  h5_filtered <- file.path(outs_dir, "filtered_feature_bc_matrix.h5")
  mtx_filtered <- file.path(outs_dir, "filtered_feature_bc_matrix")
  h5_raw <- file.path(outs_dir, "raw_feature_bc_matrix.h5")
  
  if (file.exists(h5_filtered)) {
    message("读取 filtered h5：", h5_filtered)
    counts <- Seurat::Read10X_h5(h5_filtered)
  } else if (dir.exists(mtx_filtered)) {
    message("读取 filtered mtx：", mtx_filtered)
    counts <- Seurat::Read10X(mtx_filtered)
  } else if (file.exists(h5_raw)) {
    message("读取 raw h5：", h5_raw)
    counts <- Seurat::Read10X_h5(h5_raw)
  } else {
    stop("找不到表达矩阵：", outs_dir)
  }
  
  if (is.list(counts)) {
    if ("Gene Expression" %in% names(counts)) {
      counts <- counts[["Gene Expression"]]
    } else {
      counts <- counts[[1]]
    }
  }
  
  counts <- as(counts, "dgCMatrix")
  colnames(counts) <- make_spot_id(colnames(counts), sample_id)
  
  counts
}

read_spatial_positions <- function(outs_dir, sample_id) {
  spatial_dir <- file.path(outs_dir, "spatial")
  
  f1 <- file.path(spatial_dir, "tissue_positions.csv")
  f2 <- file.path(spatial_dir, "tissue_positions_list.csv")
  
  if (file.exists(f1)) {
    pos <- read.csv(f1, stringsAsFactors = FALSE, check.names = FALSE)
  } else if (file.exists(f2)) {
    pos <- read.csv(f2, header = FALSE, stringsAsFactors = FALSE, check.names = FALSE)
    colnames(pos) <- c(
      "barcode",
      "in_tissue",
      "array_row",
      "array_col",
      "pxl_row_in_fullres",
      "pxl_col_in_fullres"
    )
  } else {
    stop("找不到空间坐标文件：", spatial_dir)
  }
  
  if (!"barcode" %in% colnames(pos)) colnames(pos)[1] <- "barcode"
  if (!"in_tissue" %in% colnames(pos)) pos$in_tissue <- 1
  if (!"array_row" %in% colnames(pos)) pos$array_row <- NA_real_
  if (!"array_col" %in% colnames(pos)) pos$array_col <- NA_real_
  
  if (!"pxl_row_in_fullres" %in% colnames(pos) ||
      !"pxl_col_in_fullres" %in% colnames(pos)) {
    if (all(c("array_row", "array_col") %in% colnames(pos))) {
      pos$pxl_row_in_fullres <- pos$array_row
      pos$pxl_col_in_fullres <- pos$array_col
    } else {
      stop("空间坐标文件缺少像素坐标：", spatial_dir)
    }
  }
  
  pos %>%
    mutate(
      spot_id = make_spot_id(barcode, sample_id),
      sample_id = sample_id,
      sample_short = ifelse(sample_id == SAMPLE_L, "L", "R"),
      barcode_core = barcode_core(barcode),
      in_tissue = as.numeric(in_tissue),
      array_row = as.numeric(array_row),
      array_col = as.numeric(array_col),
      pxl_row_in_fullres = as.numeric(pxl_row_in_fullres),
      pxl_col_in_fullres = as.numeric(pxl_col_in_fullres)
    ) %>%
    distinct(spot_id, .keep_all = TRUE)
}

read_hnas_map <- function(file, sample_short) {
  if (!file.exists(file)) stop("找不到 HNas 对照文件：", file)
  
  df <- read.csv(file, stringsAsFactors = FALSE, check.names = FALSE)
  
  if (!"Barcode" %in% colnames(df)) colnames(df)[1] <- "Barcode"
  if (!"HNas" %in% colnames(df)) colnames(df)[2] <- "HNas"
  
  df %>%
    mutate(
      sample_short = sample_short,
      barcode_core = barcode_core(Barcode),
      HNas_code = toupper(trimws(as.character(HNas))),
      circle_id = paste0(sample_short, "_", HNas_code)
    ) %>%
    filter(
      !is.na(barcode_core),
      barcode_core != "",
      !is.na(HNas_code),
      HNas_code != ""
    ) %>%
    distinct(sample_short, barcode_core, .keep_all = TRUE) %>%
    select(sample_short, barcode_core, HNas_code, circle_id)
}

read_clinical3 <- function(file, trait_name) {
  if (!file.exists(file)) {
    warning("临床文件不存在，跳过：", file)
    return(data.frame(HNas_code = character(0)))
  }
  
  df <- as.data.frame(readxl::read_excel(file, col_names = FALSE))
  
  if (ncol(df) < 3) {
    warning("临床文件列数不足，跳过：", file)
    return(data.frame(HNas_code = character(0)))
  }
  
  df <- df[, 1:3]
  colnames(df) <- c("HNas_code", "patient_id_tmp", trait_name)
  
  df %>%
    mutate(
      HNas_code = toupper(trimws(as.character(HNas_code))),
      !!trait_name := trimws(as.character(.data[[trait_name]]))
    ) %>%
    filter(
      !is.na(HNas_code),
      HNas_code != ""
    ) %>%
    select(HNas_code, all_of(trait_name)) %>%
    distinct(HNas_code, .keep_all = TRUE)
}

read_marker_modules_from_excel <- function(file) {
  if (!file.exists(file)) stop("marker 文件不存在：", file)
  
  sheets <- readxl::excel_sheets(file)
  
  all_modules <- list()
  all_meta <- data.frame()
  
  for (sh in sheets) {
    raw <- tryCatch(
      as.data.frame(readxl::read_excel(file, sheet = sh, col_names = FALSE)),
      error = function(e) NULL
    )
    
    if (is.null(raw) || nrow(raw) == 0 || ncol(raw) == 0) next
    
    raw <- raw[, colSums(!is.na(raw)) > 0, drop = FALSE]
    
    if (nrow(raw) == 0 || ncol(raw) == 0) next
    
    first_row <- as.character(unlist(raw[1, ]))
    first_row[is.na(first_row)] <- ""
    header_cols <- which(trimws(first_row) != "")
    
    if (length(header_cols) >= 2) {
      for (i in header_cols) {
        label <- trimws(as.character(raw[1, i]))
        module_id <- sanitize_name(label)
        
        genes <- raw[-1, i, drop = TRUE]
        genes <- as.character(genes)
        genes <- trimws(genes)
        genes <- genes[!is.na(genes) & genes != ""]
        genes <- unique(genes)
        
        if (length(genes) == 0) next
        
        if (module_id %in% names(all_modules)) {
          module_id <- paste0(module_id, "_", sanitize_name(sh))
        }
        
        all_modules[[module_id]] <- genes
        
        all_meta <- bind_rows(
          all_meta,
          data.frame(
            sheet = sh,
            module_id = module_id,
            module_label = label,
            input_gene = genes,
            stringsAsFactors = FALSE
          )
        )
      }
    }
  }
  
  if (length(all_modules) == 0) {
    stop("marker Excel 没有读到任何模块。请确认第一行是模块名。")
  }
  
  list(
    modules = all_modules,
    module_meta = all_meta
  )
}

get_reduced_df <- function(cds) {
  rd <- tryCatch(monocle::reducedDimS(cds), error = function(e) NULL)
  
  if (is.null(rd) && "reducedDimS" %in% slotNames(cds)) {
    rd <- slot(cds, "reducedDimS")
  }
  
  if (is.null(rd)) stop("无法提取 reducedDimS。")
  
  cell_ids <- Biobase::sampleNames(cds)
  
  if (nrow(rd) >= 2 && ncol(rd) == length(cell_ids)) {
    data.frame(
      spot_id = cell_ids,
      Component_1 = as.numeric(rd[1, ]),
      Component_2 = as.numeric(rd[2, ]),
      stringsAsFactors = FALSE
    )
  } else if (ncol(rd) >= 2 && nrow(rd) == length(cell_ids)) {
    data.frame(
      spot_id = cell_ids,
      Component_1 = as.numeric(rd[, 1]),
      Component_2 = as.numeric(rd[, 2]),
      stringsAsFactors = FALSE
    )
  } else {
    stop("reducedDimS 维度无法识别。")
  }
}

extract_edge_df_from_cds_plot <- function(cds) {
  p <- tryCatch(
    monocle::plot_cell_trajectory(cds, color_by = "State"),
    error = function(e) NULL
  )
  
  if (is.null(p)) return(data.frame())
  
  gb <- ggplot2::ggplot_build(p)
  
  segs <- lapply(gb$data, function(d) {
    if (all(c("x", "y", "xend", "yend") %in% colnames(d))) {
      d[, c("x", "y", "xend", "yend"), drop = FALSE]
    } else {
      NULL
    }
  })
  
  segs <- segs[!vapply(segs, is.null, logical(1))]
  
  if (length(segs) == 0) return(data.frame())
  
  bind_rows(segs) %>% distinct()
}

add_edge_layer <- function(p, edge_df) {
  if (is.null(edge_df) || !is.data.frame(edge_df) || nrow(edge_df) == 0) {
    return(p)
  }
  
  if (all(c("x", "y", "xend", "yend") %in% colnames(edge_df))) {
    p <- p +
      geom_segment(
        data = edge_df,

      )
  }

}

manual_find <- function(target, used_ids = character(0)) {
  lk <- module_lookup %>%
    filter(!original_module_id %in% used_ids)
  
  if (target == "Tumor_cell_expression") {
    hit <- lk %>% filter(grepl("tumor|epithelial|pithelial", norm_text))
  } else if (target == "myCAF_expression") {
    hit <- lk %>% filter(grepl("mycaf|my", norm_text) & grepl("marker|caf", norm_text))
  } else if (target == "iCAF_expression") {
    hit <- lk %>% filter(grepl("icaf|imarker|^i", norm_text))
  } else if (target == "apCAF_expression") {
    hit <- lk %>%
      filter(
        grepl("apcaf|ap", norm_text) &
          grepl("marker|caf", norm_text) &
          !grepl("derived|ecm|sasp|senescence|nosenescence|presenescence|other", norm_text)
      )
  } else if (target == "non_senescence_apCAF") {
    hit <- lk %>% filter(grepl("nosenescence|nonsenescence", norm_text))
  } else if (target == "senescence_apCAF") {
    hit <- lk %>%
      filter(
        grepl("senescence", norm_text) &
          grepl("ap|caf", norm_text) &
          !grepl("no|non|pre|ecm|sasp|derived|other", norm_text)
      )
  } else if (target == "pre_senescence_apCAF") {
    hit <- lk %>% filter(grepl("presenescence|pre", norm_text))
  } else if (target == "other_apCAF") {
    hit <- lk %>% filter(grepl("other", norm_text))
  } else if (target == "ECM_senescent_apCAF") {
    hit <- lk %>%
      filter(
        grepl("ecm", norm_text) &
          grepl("senescence|senescent", norm_text) &
          !grepl("derived", norm_text)
      )
  } else if (target == "ECM_senescent_apCAF_derived_ECM") {
    hit <- lk %>%
      filter(grepl("ecm", norm_text) & grepl("derived", norm_text))
  } else if (target == "SASP_senescent_apCAF") {
    hit <- lk %>%
      filter(
        grepl("sasp", norm_text) &
          grepl("senescence|senescent", norm_text) &
          !grepl("derived", norm_text)
      )
  } else if (target == "SASP_senescent_apCAF_derived_SASP") {
    hit <- lk %>%
      filter(grepl("sasp", norm_text) & grepl("derived", norm_text))
  } else {
    hit <- lk[0, ]
  }
  
  if (nrow(hit) == 0) return(NA_character_)
  hit$original_module_id[1]
}

cor_df <- lapply(seq_len(nrow(MODULE_DISPLAY_SPEC_OK)), function(i) {
  mid <- MODULE_DISPLAY_SPEC_OK$display_id[i]
  lab <- MODULE_DISPLAY_SPEC_OK$display_label[i]
  z_col <- paste0(mid, "__UCell_z")
  
  x <- plot_df$Pseudotime_0_1
  y <- plot_df[[z_col]]
  
  ok <- is.finite(x) & is.finite(y)
  
  if (sum(ok) < 10) {
    return(
      data.frame(
        display_order = MODULE_DISPLAY_SPEC_OK$display_order[i],
        display_id = mid,
        display_label = lab,
        n_spots = sum(ok),
        spearman_rho = NA_real_,
        spearman_p = NA_real_,
        stringsAsFactors = FALSE
      )
    )
  }
  
  sp <- suppressWarnings(cor.test(x[ok], y[ok], method = "spearman", exact = FALSE))
  
  data.frame(
    display_order = MODULE_DISPLAY_SPEC_OK$display_order[i],
    display_id = mid,
    display_label = lab,
    n_spots = sum(ok),
    spearman_rho = as.numeric(sp$estimate),
    spearman_p = sp$p.value,
    stringsAsFactors = FALSE
  )
}) %>%
  bind_rows() %>%
  mutate(
    spearman_FDR = p.adjust(spearman_p, method = "BH"),
    trend_direction = case_when(
      is.na(spearman_rho) ~ "NA",
      spearman_rho > 0 ~ "increase_along_pseudotime",
      spearman_rho < 0 ~ "decrease_along_pseudotime",
      TRUE ~ "flat"
    )
  ) %>%
  arrange(display_order)

  summarise(
    n_spots = n(),
    mean_pseudotime = mean(Pseudotime_0_1, na.rm = TRUE),
    median_pseudotime = median(Pseudotime_0_1, na.rm = TRUE),
    mean_COL1A1_score_z = mean(COL1A1_score_z, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  arrange(mean_pseudotime)

  count(State, sample_short, name = "n_spots") %>%
  group_by(State) %>%
  mutate(percent_in_state = n_spots / sum(n_spots) * 100) %>%
  ungroup()
