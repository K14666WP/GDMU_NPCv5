clean_gene_key <- function(x) {
  x <- toupper(as.character(x))
  x <- trimws(x)
  x <- gsub("\\s+", "", x)
  x <- gsub("[^A-Z0-9\\-]", "", x)
  x
}

safe_sheet <- function(x) {
  x <- gsub("[\\[\\]\\*\\?/\\\\:]", "_", x)
  x <- substr(x, 1, 31)
  ifelse(x == "", "sheet", x)
}

safe_rescale <- function(x, to = c(1, 8)) {
  x <- as.numeric(x)
  out <- rep(mean(to), length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  
  if (length(unique(x[ok])) == 1) {
    out[ok] <- mean(to)
  } else {
    out[ok] <- scales::rescale(x[ok], to = to)
  }
  
  out
}

save_pdf_png <- function(p, prefix, outdir, width = 10, height = 8, dpi = 320) {
  dir.create(outdir, recursive = TRUE, showWarnings = FALSE)
  
  pdf_file <- file.path(outdir, paste0(prefix, ".pdf"))
  png_file <- file.path(outdir, paste0(prefix, ".png"))
  
  ggplot2::ggsave(
    filename = pdf_file,
    plot = p,
    width = width,
    height = height,
    bg = "white",
    limitsize = FALSE
  )
  
  ggplot2::ggsave(
    filename = png_file,
    plot = p,
    width = width,
    height = height,
    dpi = dpi,
    bg = "white",
    limitsize = FALSE
  )
  
  message("[PDF] ", pdf_file)
  message("[PNG] ", png_file)
  
  invisible(list(pdf = pdf_file, png = png_file))
}

init_STRINGdb <- function() {
  last_error <- NULL
  
  for (v in STRING_VERSION_CANDIDATES) {
    message("尝试初始化 STRINGdb version = ", v)
    
    obj <- tryCatch(
      {
        STRINGdb::STRINGdb$new(
          version = v,
          species = STRING_SPECIES,
          score_threshold = 0,
          input_directory = DIR_CACHE
        )
      },
      error = function(e) {
        last_error <<- e$message
        NULL
      }
    )
    
    if (!is.null(obj)) {
      message("STRINGdb 初始化成功，version = ", v)
      return(list(db = obj, version = v))
    }
  }
  
  stop(
    "STRINGdb 初始化失败。最后错误信息：",
    last_error,
    "\n请检查网络连接或 STRINGdb 安装。"
  )
}

find_score_col <- function(df) {
  cand <- c(
    "combined_score",
    "score",
    "combined.score",
    "combinedscore"
  )
  
  hit <- cand[cand %in% colnames(df)]
  
  if (length(hit) > 0) {
    return(hit[1])
  }
  
  stop("STRING 互作结果中找不到 combined_score / score 列。")
}

## 第一次下载失败后，可能会残留不完整文件；这里先清理可疑的临时/残缺文件
clean_incomplete_string_cache <- function(cache_dir) {
  if (!dir.exists(cache_dir)) return(invisible(NULL))
  
  candidate_files <- list.files(
    cache_dir,
    pattern = "protein\\.links|\\.tmp|tmp",
    recursive = TRUE,
    full.names = TRUE,
    ignore.case = TRUE
  )
  
  if (length(candidate_files) == 0) return(invisible(NULL))
  
  info <- file.info(candidate_files)
  bad_files <- rownames(info)[
    !is.na(info$size) &
      info$size > 0 &
      info$size < 80000000
  ]
  
  if (length(bad_files) > 0) {
    message("删除可能未下载完整的 STRING 缓存文件数量：", length(bad_files))
    unlink(bad_files, force = TRUE)
  }
  
  invisible(NULL)
}

get_string_interactions_retry <- function(string_db, string_ids, max_try = 5) {
  last_error <- NULL
  
  for (ii in seq_len(max_try)) {
    message("STRING get_interactions 尝试第 ", ii, " / ", max_try, " 次")
    
    ppi_try <- tryCatch(
      {
        string_db$get_interactions(string_ids)
      },
      error = function(e) {
        last_error <<- e$message
        NULL
      }
    )
    
    if (!is.null(ppi_try) && is.data.frame(ppi_try) && nrow(ppi_try) > 0) {
      message("STRING get_interactions 成功。")
      return(ppi_try)
    }
    
    message("本次 STRING get_interactions 失败。")
    
    if (!is.null(last_error)) {
      message("错误信息：", last_error)
    }
    
    clean_incomplete_string_cache(DIR_CACHE)
    
    wait_sec <- min(60, ii * 15)
    message("等待 ", wait_sec, " 秒后重试。")
    Sys.sleep(wait_sec)
  }
  
  stop(
    "STRINGdb get_interactions 连续失败。最后错误信息：",
    last_error,
    "\n建议检查服务器网络，或手动下载 STRING protein.links 文件。"
  )
}

  d = edges_for_graph,
  directed = FALSE,
  vertices = nodes_base
)

V(g)$weighted_degree <- igraph::strength(g, weights = E(g)$weight_score)
V(g)$betweenness <- igraph::betweenness(
  g,
  weights = E(g)$distance_weight,
  normalized = TRUE
)
V(g)$closeness <- suppressWarnings(
  igraph::closeness(
    g,
    weights = E(g)$distance_weight,
    normalized = TRUE
  )
)
V(g)$component_id <- comp$membership
V(g)$component_size <- comp$csize[comp$membership]
V(g)$in_largest_component <- V(g)$component_id == which.max(comp$csize)

nodes_metrics <- igraph::as_data_frame(g, what = "vertices") %>%
  dplyr::rename(STRING_id = name) %>%
  dplyr::mutate(
    degree_rank = rank(-degree, ties.method = "min"),
    weighted_degree_rank = rank(-weighted_degree, ties.method = "min"),

match_gene_in_semicolon_vector <- function(gene, vec) {
  gene <- clean_gene_key(gene)
  vec <- as.character(vec)
  vec <- vec[!is.na(vec) & vec != ""]
  
  if (length(vec) == 0 || is.na(gene) || gene == "") {
    return(FALSE)
  }
  
  any(
    vapply(
      strsplit(vec, ";", fixed = TRUE),
      function(x) {
        gene %in% clean_gene_key(x)
      },
      logical(1)
    )
  )
}

  dplyr::rowwise() %>%
  dplyr::mutate(
    present_in_intersection = gene_for_matching %in% common_genes,
    
    mapped_to_STRING = gene_for_matching %in% clean_gene_key(mapped_ok$gene) |
      gene_for_matching %in% clean_gene_key(mapped_ok$preferred_name),
    
    in_PPI_network = gene_for_matching %in% clean_gene_key(nodes_metrics$gene_clean) |
      match_gene_in_semicolon_vector(
        gene = gene_for_matching,
        vec = nodes_metrics$input_gene_clean_all
      ) |
      match_gene_in_semicolon_vector(
        gene = gene_for_matching,
        vec = nodes_metrics$key_gene_symbol_matched
      ) |
      match_gene_in_semicolon_vector(
        gene = gene_for_matching,
        vec = nodes_metrics$key_gene_input_matched
      )
  ) %>%
  dplyr::ungroup() %>%
  dplyr::arrange(
    dplyr::desc(in_PPI_network),
    dplyr::desc(mapped_to_STRING),
    dplyr::desc(present_in_intersection),
    key_gene_input,
    gene_for_matching
