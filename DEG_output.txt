find_first_col <- function(df, candidates, required = TRUE, desc = "") {
  hit <- candidates[candidates %in% colnames(df)]
  
  if (length(hit) == 0) {
    if (required) {
      stop(
        "缺少必要列：",
        desc,
        "\n候选列名：",
        paste(candidates, collapse = ", "),
        "\n当前列名：",
        paste(colnames(df), collapse = ", ")
      )
    } else {
      return(NA_character_)
    }
  }
  
  hit[1]
}

clean_gene <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("\\..*$", "", x)
  x <- toupper(x)
  x[x %in% c("", "NA", "NAN", "NULL")] <- NA_character_
  x
}

make_safe_name <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("[\r\n\t]+", " ", x)
  x <- gsub("\\s+", "_", x)
  x <- gsub("[/\\\\:*?\"<>|]", "_", x, perl = TRUE)
  x <- gsub("_+", "_", x)
  x <- gsub("^_|_$", "", x)
  ifelse(is.na(x) | x == "", "module", x)
}

safe_sheet <- function(x) {
  x <- as.character(x)
  x <- gsub("[\\[\\]\\*\\?/\\\\:]", "_", x)
  x <- gsub("^'|'$", "", x)
  x <- substr(x, 1, 31)
  ifelse(x == "", "sheet", x)
}

derive_direction <- function(comparison, avg_diff) {
  comparison <- as.character(comparison)
  avg_diff <- as.numeric(avg_diff)
  
  dplyr::case_when(
    comparison == "Hotspot_vs_Near" & is.finite(avg_diff) & avg_diff > 0 ~ "Hotspot_up",
    comparison == "Hotspot_vs_Near" & is.finite(avg_diff) & avg_diff < 0 ~ "Near_up",
    comparison == "Near_vs_Far" & is.finite(avg_diff) & avg_diff > 0 ~ "Near_up",
    comparison == "Near_vs_Far" & is.finite(avg_diff) & avg_diff < 0 ~ "Far_up",
    TRUE ~ "unknown"
  )
}

parse_comparison <- function(file) {
  bn <- basename(file)
  
  dplyr::case_when(
    grepl("Hotspot_vs_Near", bn, ignore.case = TRUE) ~ "Hotspot_vs_Near",
    grepl("Near_vs_Far", bn, ignore.case = TRUE) ~ "Near_vs_Far",
    TRUE ~ NA_character_
  )
}

parse_section <- function(file) {
  bn <- basename(file)
  
  dplyr::case_when(
    grepl("_DEG_L\\.csv$", bn, ignore.case = TRUE) ~ "L",
    grepl("_DEG_R\\.csv$", bn, ignore.case = TRUE) ~ "R",
    grepl("_L\\.csv$", bn, ignore.case = TRUE) & !grepl("LR_combined", bn, ignore.case = TRUE) ~ "L",
    grepl("_R\\.csv$", bn, ignore.case = TRUE) & !grepl("LR_combined", bn, ignore.case = TRUE) ~ "R",
    TRUE ~ NA_character_
  )
}

parse_module_from_path <- function(file) {
  ## 典型结构：
  ## <root>/<module>/01_tables/06_xxx_DEG_L.csv
  parent1 <- basename(dirname(file))
  parent2 <- basename(dirname(dirname(file)))
  
  if (parent1 == "01_tables") {
    return(parent2)
  }
  
  parent2
}

pick_best_record <- function(df) {
  if (nrow(df) == 0) return(df)
  
  df %>%
    dplyr::arrange(p_value, p_adj_BH, dplyr::desc(abs(avg_diff))) %>%
    dplyr::slice(1)
}

read_one_deg <- function(file, module_label, section, comparison) {
  
  df <- read.csv(
    file,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  gene_col <- find_first_col(
    df,
    c("gene", "Gene", "genes", "SYMBOL", "symbol"),
    required = TRUE,
    desc = "gene"
  )
  
  p_col <- find_first_col(
    df,
    c("p_wilcox", "p_value", "pvalue", "p_val", "p", "P.Value", "p.value"),
    required = TRUE,
    desc = "p value"
  )
  
  padj_col <- find_first_col(
    df,
    c("p_adj_BH", "p.adjust", "p_adj", "padj", "FDR", "fdr", "adj.P.Val"),
    required = FALSE
  )
  
  avg_col <- find_first_col(
    df,
    c(
      "avg_diff_group_a_minus_group_b",
      "avg_diff_near_minus_far",
      "avg_diff",
      "avg_log2FC",
      "logFC",
      "log2FC",
      "diff"
    ),
    required = FALSE
  )
  
  dir_col <- find_first_col(
    df,
    c("direction", "trend", "Direction", "regulation"),
    required = FALSE
  )
  
  out <- df %>%
    dplyr::mutate(
      module_label = module_label,
      section = section,
      comparison = comparison,
      gene = clean_gene(.data[[gene_col]]),
      p_value = as.numeric(.data[[p_col]]),
      p_col_used = p_col,
      p_adj_BH = if (!is.na(padj_col)) as.numeric(.data[[padj_col]]) else NA_real_,
      avg_diff = if (!is.na(avg_col)) as.numeric(.data[[avg_col]]) else NA_real_,
      direction_from_file = if (!is.na(dir_col)) as.character(.data[[dir_col]]) else NA_character_,
      direction_by_avgdiff = derive_direction(comparison, avg_diff),
      source_file = file
    ) %>%
    dplyr::filter(
      !is.na(gene),
      gene != "",
      is.finite(p_value)
    ) %>%
    dplyr::select(
      module_label,
      section,
      comparison,
      gene,
      p_value,
      p_col_used,
      p_adj_BH,
      avg_diff,
      direction_by_avgdiff,
      direction_from_file,
      source_file,
      dplyr::everything()
    )
  
  out
}

summarise_source_by_gene <- function(sig_df) {
  
  if (nrow(sig_df) == 0) {
    return(data.frame())
  }
  
  best_df <- sig_df %>%
    dplyr::group_by(gene, section, comparison) %>%
    dplyr::group_modify(~ pick_best_record(.x)) %>%
    dplyr::ungroup()
  
  presence <- best_df %>%
    dplyr::mutate(
      presence_key = paste(section, comparison, "sig", sep = "_"),
      present = TRUE
    ) %>%
    dplyr::select(gene, presence_key, present) %>%
    tidyr::pivot_wider(
      names_from = presence_key,
      values_from = present,
      values_fill = FALSE
    )
  
  p_wide <- best_df %>%
    dplyr::mutate(stat_key = paste(section, comparison, "p", sep = "_")) %>%
    dplyr::select(gene, stat_key, p_value) %>%
    tidyr::pivot_wider(
      names_from = stat_key,
      values_from = p_value
    )
  
  padj_wide <- best_df %>%
    dplyr::mutate(stat_key = paste(section, comparison, "padj", sep = "_")) %>%
    dplyr::select(gene, stat_key, p_adj_BH) %>%
    tidyr::pivot_wider(
      names_from = stat_key,
      values_from = p_adj_BH
    )
  
  avg_wide <- best_df %>%
    dplyr::mutate(stat_key = paste(section, comparison, "avg_diff", sep = "_")) %>%
    dplyr::select(gene, stat_key, avg_diff) %>%
    tidyr::pivot_wider(
      names_from = stat_key,
      values_from = avg_diff
    )
  
  dir_wide <- best_df %>%
    dplyr::mutate(stat_key = paste(section, comparison, "direction", sep = "_")) %>%
    dplyr::select(gene, stat_key, direction_by_avgdiff) %>%
    tidyr::pivot_wider(
      names_from = stat_key,
      values_from = direction_by_avgdiff
    )
  
  section_summary <- best_df %>%
    dplyr::group_by(gene, section) %>%
    dplyr::summarise(
      source_comparisons = paste(unique(comparison), collapse = ";"),
      n_source_comparisons = dplyr::n_distinct(comparison),
      min_p_value = min(p_value, na.rm = TRUE),
      directions = paste(unique(direction_by_avgdiff), collapse = ";"),
      .groups = "drop"
    ) %>%
    tidyr::pivot_wider(
      id_cols = gene,
      names_from = section,
      values_from = c(
        source_comparisons,
        n_source_comparisons,
        min_p_value,
        directions
      ),
      names_glue = "{section}_{.value}"
    )
  
  out <- presence %>%
    dplyr::full_join(p_wide, by = "gene") %>%
    dplyr::full_join(padj_wide, by = "gene") %>%
    dplyr::full_join(avg_wide, by = "gene") %>%
    dplyr::full_join(dir_wide, by = "gene") %>%
    dplyr::full_join(section_summary, by = "gene")
  
  needed_presence <- c(
    "L_Hotspot_vs_Near_sig",
    "L_Near_vs_Far_sig",
    "R_Hotspot_vs_Near_sig",
    "R_Near_vs_Far_sig"
  )
  
  for (cc in needed_presence) {
    if (!cc %in% colnames(out)) {
      out[[cc]] <- FALSE
    }
  }
  
  out %>%
    dplyr::mutate(
      L_any_sig = L_Hotspot_vs_Near_sig | L_Near_vs_Far_sig,
      R_any_sig = R_Hotspot_vs_Near_sig | R_Near_vs_Far_sig
    )
}

run_one_module <- function(module_label) {
  
  message("============================================================")
  message("整理模块：", module_label)
  message("============================================================")
  
  module_dir_name <- make_safe_name(module_label)
  out_module_dir <- file.path(DIR_MODULE, module_dir_name)
  out_tab <- file.path(out_module_dir, "01_tables")
  out_txt <- file.path(out_module_dir, "02_gene_lists_txt")
  out_xlsx <- file.path(out_module_dir, "03_excel")
  
  dir.create(out_module_dir, recursive = TRUE, showWarnings = FALSE)
  dir.create(out_tab, recursive = TRUE, showWarnings = FALSE)
  dir.create(out_txt, recursive = TRUE, showWarnings = FALSE)
  dir.create(out_xlsx, recursive = TRUE, showWarnings = FALSE)
  
  sig_m <- deg_sig %>%
    dplyr::filter(module_label == !!module_label)
  
  all_m <- deg_all %>%
    dplyr::filter(module_label == !!module_label)
  
  L_union <- sig_m %>%
    dplyr::filter(section == "L") %>%
    dplyr::pull(gene) %>%
    unique() %>%
    sort()
  
  R_union <- sig_m %>%
    dplyr::filter(section == "R") %>%
    dplyr::pull(gene) %>%
    unique() %>%
    sort()
  
  LR_intersection <- intersect(L_union, R_union) %>%
    sort()
  
  source_detail_all <- summarise_source_by_gene(sig_m)
  
  L_union_df <- source_detail_all %>%
    dplyr::filter(gene %in% L_union) %>%
    dplyr::arrange(gene)
  
  R_union_df <- source_detail_all %>%
    dplyr::filter(gene %in% R_union) %>%
    dplyr::arrange(gene)
  
  intersection_detail <- source_detail_all %>%
    dplyr::filter(gene %in% LR_intersection) %>%
    dplyr::mutate(
      module_label = module_label,
      gene_rank = dplyr::row_number(),
      .before = 1
    ) %>%
    dplyr::arrange(gene)
  
  intersection_simple <- data.frame(
    module_label = module_label,
    gene = LR_intersection,
    stringsAsFactors = FALSE
  )
  
  summary_df <- data.frame(
    module_label = module_label,
    module_output_dir = out_module_dir,
    n_L_Hotspot_vs_Near_p005 = sig_m %>%
      dplyr::filter(section == "L", comparison == "Hotspot_vs_Near") %>%
      dplyr::pull(gene) %>%
      unique() %>%
      length(),
    n_L_Near_vs_Far_p005 = sig_m %>%
      dplyr::filter(section == "L", comparison == "Near_vs_Far") %>%
      dplyr::pull(gene) %>%
      unique() %>%
      length(),
    n_L_union_p005 = length(L_union),
    n_R_Hotspot_vs_Near_p005 = sig_m %>%
      dplyr::filter(section == "R", comparison == "Hotspot_vs_Near") %>%
      dplyr::pull(gene) %>%
      unique() %>%
      length(),
    n_R_Near_vs_Far_p005 = sig_m %>%
      dplyr::filter(section == "R", comparison == "Near_vs_Far") %>%
      dplyr::pull(gene) %>%
      unique() %>%
      length(),
    n_R_union_p005 = length(R_union),
    n_L_union_intersect_R_union = length(LR_intersection),
    p_cutoff = P_CUTOFF,
    stringsAsFactors = FALSE
  )
  
  write.csv(
    summary_df,
    file.path(out_tab, "00_module_L_R_union_intersection_summary.csv"),
    row.names = FALSE
  )
  
  write.csv(
    L_union_df,
    file.path(out_tab, "01_L_two_comparisons_union_p005_detail.csv"),
    row.names = FALSE
  )
  
  write.csv(
    R_union_df,
    file.path(out_tab, "02_R_two_comparisons_union_p005_detail.csv"),
    row.names = FALSE
  )
  
  write.csv(
    intersection_detail,
    file.path(out_tab, "03_L_union_intersect_R_union_p005_detail.csv"),
    row.names = FALSE
  )
  
  write.csv(
    intersection_simple,
    file.path(out_tab, "04_L_union_intersect_R_union_p005_gene_list_simple.csv"),
    row.names = FALSE
  )
  
  write.csv(
    sig_m,
    file.path(out_tab, "05_all_source_DEG_records_p005_this_module.csv"),
    row.names = FALSE
  )
  
  write.csv(
    all_m,
    file.path(out_tab, "06_all_source_DEG_records_raw_this_module.csv"),
    row.names = FALSE
  )
  
  write.table(
    LR_intersection,
    file.path(out_txt, "L_union_intersect_R_union_p005_genes.txt"),
    quote = FALSE,
    row.names = FALSE,
    col.names = FALSE
  )
  
  write.table(
    L_union,
    file.path(out_txt, "L_two_comparisons_union_p005_genes.txt"),
    quote = FALSE,
    row.names = FALSE,
    col.names = FALSE
  )
  
  write.table(
    R_union,
    file.path(out_txt, "R_two_comparisons_union_p005_genes.txt"),
    quote = FALSE,
    row.names = FALSE,
    col.names = FALSE
  )
  
  xlsx_file <- file.path(
    out_xlsx,
    paste0(module_dir_name, "_L_R_union_intersection_p005.xlsx")
  )
  
  openxlsx::write.xlsx(
    list(
      summary = summary_df,
      L_union = L_union_df,
      R_union = R_union_df,
      LR_intersection = intersection_detail,
      LR_gene_list = intersection_simple,
      source_p005 = sig_m
    ),
    file = xlsx_file,
    overwrite = TRUE
  )
  
  message("L union: ", length(L_union))
  message("R union: ", length(R_union))
  message("L ∩ R : ", length(LR_intersection))
  message("输出：", out_module_dir)
  
  list(
    summary = summary_df,
    L_union = L_union_df,
    R_union = R_union_df,
    intersection_detail = intersection_detail,
    intersection_simple = intersection_simple,
    source_p005 = sig_m,
    module_output_dir = out_module_dir
  )
}
