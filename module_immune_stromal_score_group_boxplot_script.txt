safe_z <- function(x) {
  x <- as.numeric(x)
  out <- rep(NA_real_, length(x))
  ok <- is.finite(x)
  
  if (sum(ok) == 0) return(out)
  if (sum(ok) < 2) {
    out[ok] <- 0
    return(out)
  }
  
  m <- mean(x[ok], na.rm = TRUE)
  s <- sd(x[ok], na.rm = TRUE)
  
  if (!is.finite(s) || s == 0) {
    out[ok] <- 0
  } else {
    out[ok] <- (x[ok] - m) / s
  }
  
  out
}

weighted_mean_safe <- function(x, w = NULL) {
  x <- as.numeric(x)
  
  if (is.null(w)) {
    w <- rep(1, length(x))
  }
  
  w <- as.numeric(w)
  
  ok <- is.finite(x) & is.finite(w) & w > 0
  
  if (sum(ok) == 0) return(NA_real_)
  
  sum(x[ok] * w[ok], na.rm = TRUE) / sum(w[ok], na.rm = TRUE)
}

norm_key <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- tolower(x)
  x <- gsub("[[:space:][:punct:]_\\-]+", "", x, perl = TRUE)
  x
}

sanitize_filename <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- iconv(x, from = "", to = "ASCII//TRANSLIT", sub = "_")
  x <- gsub("[\r\n\t]+", " ", x)
  x <- gsub("\\s+", "_", x)
  x <- gsub("[/\\\\:*?\"<>|]", "_", x, perl = TRUE)
  x <- gsub("[^A-Za-z0-9_.-]+", "_", x, perl = TRUE)
  x <- gsub("_+", "_", x)
  x <- gsub("^[_\\.]+|[_\\.]+$", "", x)
  ifelse(is.na(x) | x == "", "unknown", x)
}

normalize_group <- function(x) {
  y <- tolower(trimws(as.character(x)))
  y <- gsub("[[:space:]_\\-]+", "", y)
  
  dplyr::case_when(
    y %in% c("hotspot", "hot", "core") ~ "Hotspot",
    y %in% c("near") ~ "Near",
    y %in% c("middle", "mid") ~ "Middle",
    y %in% c("far") ~ "Far",
    TRUE ~ NA_character_
  )
}

normalize_section <- function(x) {
  y <- as.character(x)
  k <- norm_key(y)
  
  dplyr::case_when(
    k %in% c("lrcombined", "lr", "combined") ~ "LR_combined",
    grepl("lrcombined", k) ~ "LR_combined",
    grepl("combined", k) ~ "LR_combined",
    k == "l" ~ "L",
    k == "r" ~ "R",
    grepl("l$", k) & !grepl("lr|combined", k) ~ "L",
    grepl("r$", k) & !grepl("lr|combined", k) ~ "R",
    TRUE ~ y
  )
}

clean_hnas <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("^L_", "", x, ignore.case = TRUE)
  x <- gsub("^R_", "", x, ignore.case = TRUE)
  x <- gsub("^HNASN091SU01_L_", "", x, ignore.case = TRUE)
  x <- gsub("^HNASN091SU01_R_", "", x, ignore.case = TRUE)
  x <- gsub("^HNas_N091Su01_L_", "", x, ignore.case = TRUE)
  x <- gsub("^HNas_N091Su01_R_", "", x, ignore.case = TRUE)
  x <- gsub("\\.0$", "", x)
  x
}

clean_spot_id <- function(x) {
  x <- as.character(x)
  x <- trimws(x)
  x <- gsub("\"", "", x)
  x <- gsub("'", "", x)
  x <- gsub("^\\s+|\\s+$", "", x)
  x
}

fix_module_name <- function(x) {
  x0 <- as.character(x)
  k <- norm_key(x0)
  
  out <- rep(NA_character_, length(k))
  
  out[grepl("ecmapcafmarkerassociatedecm", k)] <- "REMOVE_ECM_marker_associated_ECM"
  out[grepl("saspapcafmarkerassociatedsasp", k)] <- "REMOVE_SASP_marker_associated_SASP"
  
  out[is.na(out) & grepl("^otherapcaf$", k)] <- "no-senescence apCAF"
  out[is.na(out) & grepl("otherapcaf", k)] <- "no-senescence apCAF"
  out[is.na(out) & grepl("nosenescenceapcaf", k)] <- "no-senescence apCAF"
  
  out[is.na(out) & grepl("presenescenceapcaf", k)] <- "pre-senescence apCAF"
  out[is.na(out) & grepl("presenescentapcaf", k)] <- "pre-senescence apCAF"
  out[is.na(out) & grepl("presenescence", k)] <- "pre-senescence apCAF"
  out[is.na(out) & grepl("presenescent", k)] <- "pre-senescence apCAF"
  
  out[is.na(out) & grepl("ecmsenescenceapcaf", k)] <- "ECM senescence apCAF"
  out[is.na(out) & grepl("ecmsenescentapcaf", k)] <- "ECM senescence apCAF"
  
  out[is.na(out) & grepl("saspsenescenceapcaf", k)] <- "SASP senescence apCAF"
  out[is.na(out) & grepl("saspsenescentapcaf", k)] <- "SASP senescence apCAF"
  
  out[is.na(out) & grepl("^senescenceapcaf$", k)] <- "senescence apCAF"
  out[is.na(out) & grepl("^senescentapcaf$", k)] <- "senescence apCAF"
  
  out[is.na(out)] <- x0[is.na(out)]
  
  out
}

module_source_priority <- function(module_label_original) {
  k <- norm_key(module_label_original)
  
  dplyr::case_when(
    grepl("otherapcaf", k) ~ 1L,
    grepl("nosenescenceapcaf", k) ~ 2L,
    TRUE ~ 1L
  )
}

find_first_col <- function(df, candidates) {
  hit <- candidates[candidates %in% colnames(df)]
  if (length(hit) == 0) return(NA_character_)
  hit[1]
}

count_spots_in_string <- function(x) {
  x <- as.character(x)
  x <- gsub("\\[|\\]|\"", "", x)
  x <- gsub("'", "", x)
  out <- unlist(strsplit(x, split = "[,;|[:space:]]+"))
  out <- clean_spot_id(out)
  out <- out[!is.na(out) & out != ""]
  length(unique(out))
}

pick_score_col <- function(df) {
  preferred <- c(
    "module_score",
    "marker_score",
    "module_score_z",
    "marker_score_z",
    "score",
    "spot_score",
    "UCell_score",
    "ucell_score"
  )
  
  hit <- preferred[preferred %in% colnames(df)]
  
  if (length(hit) > 0) return(hit[1])
  
  cn <- colnames(df)
  
  cand <- cn[
    grepl("module.*score|marker.*score|ucell.*score|score", cn, ignore.case = TRUE) &
      !grepl("xcell|distance|near|far|hotspot|group|mean|median|pct|count|n_", cn, ignore.case = TRUE)
  ]
  
  if (length(cand) > 0) return(cand[1])
  
  NA_character_
}

read_one_spot_circle <- function(f) {
  
  df <- read.csv(
    f,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  module_col <- find_first_col(df, c("object_label", "module_label", "object_id", "module"))
  section_col <- find_first_col(df, c("section", "sample_short"))
  spot_col <- find_first_col(df, c("spot_id", "barcode"))
  score_col <- pick_score_col(df)
  circle_col <- find_first_col(df, c("circle_id", "pseudo_circle_id"))
  hnas_col <- find_first_col(df, c("HNas_code", "hnas_code", "HNas"))
  
  if (is.na(module_col)) stop("spot 文件缺少模块列：", f)
  if (is.na(section_col)) stop("spot 文件缺少 section/sample_short：", f)
  if (is.na(spot_col)) stop("spot 文件缺少 spot_id/barcode：", f)
  if (is.na(score_col)) stop("spot 文件缺少 module score：", f)
  if (is.na(circle_col) && is.na(hnas_col)) stop("spot 文件缺少 circle_id/HNas_code：", f)
  
  module_label_original <- unique(na.omit(as.character(df[[module_col]])))[1]
  module_label <- fix_module_name(module_label_original)
  
  if (grepl("^REMOVE_", module_label)) {
    return(data.frame())
  }
  
  if (!module_label %in% MODULE_ORDER_FINAL) {
    return(data.frame())
  }
  
  out_spot <- df %>%
    dplyr::transmute(
      module_label_original = module_label_original,
      module_label = module_label,
      module_key = norm_key(module_label),
      module_source_priority = module_source_priority(module_label_original),
      section = normalize_section(.data[[section_col]]),
      spot_id = clean_spot_id(.data[[spot_col]]),
      module_score = as.numeric(.data[[score_col]]),
      circle_id = if (!is.na(circle_col)) as.character(.data[[circle_col]]) else NA_character_,
      HNas_code = if (!is.na(hnas_col)) as.character(.data[[hnas_col]]) else circle_id,
      HNas_code_clean = clean_hnas(HNas_code),
      circle_id_clean = clean_hnas(circle_id),
      source_file = f
    ) %>%
    dplyr::filter(
      section %in% c("L", "R"),
      is.finite(module_score),
      !is.na(spot_id),
      spot_id != "",
      !is.na(HNas_code_clean),
      HNas_code_clean != ""
    ) %>%
    dplyr::distinct(section, spot_id, .keep_all = TRUE)
  
  out_lr <- out_spot %>%
    dplyr::group_by(
      module_label_original,
      module_label,
      module_key,
      module_source_priority,
      section,
      HNas_code_clean
    ) %>%
    dplyr::summarise(
      mean_module_score = mean(module_score, na.rm = TRUE),
      median_module_score = median(module_score, na.rm = TRUE),
      n_spots_module = dplyr::n(),
      .groups = "drop"
    )
  
  out_lrc <- out_spot %>%
    dplyr::mutate(section = "LR_combined") %>%
    dplyr::group_by(
      module_label_original,
      module_label,
      module_key,
      module_source_priority,
      section,
      HNas_code_clean
    ) %>%
    dplyr::summarise(
      mean_module_score = mean(module_score, na.rm = TRUE),
      median_module_score = median(module_score, na.rm = TRUE),
      n_spots_module = dplyr::n(),
      .groups = "drop"
    )
  
  dplyr::bind_rows(out_lr, out_lrc)
}

read_one_sample_info <- function(f, module_label_from_file = NA_character_) {
  
  df <- read.csv(
    f,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  pseudo_col <- find_first_col(df, c("pseudo_sample_id", "sample_id"))
  section_col <- find_first_col(df, c("section", "sample_short"))
  group_col <- find_first_col(df, c("xcell_group", "group", "distance_group", "group_label"))
  circle_col <- find_first_col(df, c("pseudo_circle_id", "circle_id", "HNas_code", "hnas_code"))
  hnas_col <- find_first_col(df, c("HNas_code", "hnas_code", "HNas"))
  spot_ids_col <- find_first_col(df, c("spot_ids", "spots", "spot_id_list", "barcodes", "barcode_list"))
  n_col <- find_first_col(df, c("n_spots", "spot_count", "n_cells", "nSpot", "n"))
  
  if (is.na(pseudo_col)) stop("sample_info 缺少 pseudo_sample_id：", f)
  if (is.na(section_col)) stop("sample_info 缺少 section：", f)
  if (is.na(group_col)) stop("sample_info 缺少 xcell_group：", f)
  
  ## 这里不用 case_when，避免标量条件 warning
  if (!is.na(n_col)) {
    n_spots_vec <- as.numeric(df[[n_col]])
  } else if (!is.na(spot_ids_col)) {
    n_spots_vec <- as.numeric(
      vapply(
        as.character(df[[spot_ids_col]]),
        count_spots_in_string,
        numeric(1)
      )
    )
  } else {
    n_spots_vec <- rep(1, nrow(df))
  }
  
  out <- df %>%
    dplyr::mutate(
      pseudo_sample_id = as.character(.data[[pseudo_col]]),
      section = normalize_section(.data[[section_col]]),
      xcell_group = normalize_group(.data[[group_col]]),
      pseudo_circle_id = if (!is.na(circle_col)) as.character(.data[[circle_col]]) else NA_character_,
      HNas_code = if (!is.na(hnas_col)) as.character(.data[[hnas_col]]) else pseudo_circle_id,
      HNas_code_clean = clean_hnas(HNas_code),
      pseudo_circle_id_clean = clean_hnas(pseudo_circle_id),
      n_spots_pseudo = n_spots_vec,
      source_sample_info_file = f
    ) %>%
    dplyr::filter(
      !is.na(pseudo_sample_id),
      pseudo_sample_id != "",
      section %in% c("LR_combined", "L", "R"),
      xcell_group %in% GROUP_ORDER,
      !is.na(HNas_code_clean),
      HNas_code_clean != ""
    ) %>%
    dplyr::select(
      pseudo_sample_id,
      section,
      xcell_group,
      pseudo_circle_id,
      pseudo_circle_id_clean,
      HNas_code,
      HNas_code_clean,
      n_spots_pseudo,
      source_sample_info_file
    ) %>%
    dplyr::distinct(pseudo_sample_id, .keep_all = TRUE)
  
  out
}


############################################################
## 修正版 read_one_xcell_long_raw
## 修复 left_join 后 section / xcell_group / HNas_code_clean 重名冲突
## 强制只保留 64 类 xCell，排除 ImmuneScore / StromaScore
############################################################

read_one_xcell_long_raw <- function(xcell_file, sample_info_file, spot_file) {
  
  ##########################################################
  ## 1. 从 spot 文件读模块名
  ##########################################################
  
  spot_df <- read.csv(
    spot_file,
    stringsAsFactors = FALSE,
    check.names = FALSE,
    nrows = 10
  )
  
  module_col <- find_first_col(
    spot_df,
    c("object_label", "module_label", "object_id", "module")
  )
  
  module_label_original <- if (!is.na(module_col)) {
    unique(na.omit(as.character(spot_df[[module_col]])))[1]
  } else {
    basename(dirname(dirname(spot_file)))
  }
  
  module_label <- fix_module_name(module_label_original)
  
  if (grepl("^REMOVE_", module_label) || !module_label %in% MODULE_ORDER_FINAL) {
    return(data.frame())
  }
  
  ##########################################################
  ## 2. 读取 sample_info
  ##########################################################
  
  sample_meta <- read_one_sample_info(sample_info_file) %>%
    dplyr::rename(
      section_meta = section,
      xcell_group_meta = xcell_group,
      pseudo_circle_id_meta = pseudo_circle_id,
      pseudo_circle_id_clean_meta = pseudo_circle_id_clean,
      HNas_code_meta = HNas_code,
      HNas_code_clean_meta = HNas_code_clean,
      n_spots_pseudo_meta = n_spots_pseudo
    )
  
  ##########################################################
  ## 3. 读取 xCell long
  ##########################################################
  
  xlong <- read.csv(
    xcell_file,
    stringsAsFactors = FALSE,
    check.names = FALSE
  )
  
  pseudo_col <- find_first_col(xlong, c("pseudo_sample_id", "sample_id"))
  section_col <- find_first_col(xlong, c("section", "sample_short"))
  group_col <- find_first_col(xlong, c("xcell_group", "group", "distance_group", "group_label"))
  circle_col <- find_first_col(xlong, c("pseudo_circle_id", "circle_id", "HNas_code", "hnas_code"))
  hnas_col <- find_first_col(xlong, c("HNas_code", "hnas_code", "HNas"))
  celltype_col <- find_first_col(xlong, c("xcell_celltype", "cell_type", "celltype"))
  score_col <- find_first_col(xlong, c("xcell_score", "score", "adjustedScore", "adjusted_score", "enrichment_score", "raw_score"))
  
  if (is.na(pseudo_col)) stop("xCell long 缺少 pseudo_sample_id：", xcell_file)
  if (is.na(celltype_col)) stop("xCell long 缺少 xcell_celltype：", xcell_file)
  if (is.na(score_col)) stop("xCell long 缺少 xcell_score：", xcell_file)
  
  ## 先只生成不会和 sample_meta 冲突的 *_x 列
  xraw0 <- xlong %>%
    dplyr::transmute(
      pseudo_sample_id = as.character(.data[[pseudo_col]]),
      section_x = if (!is.na(section_col)) normalize_section(.data[[section_col]]) else NA_character_,
      xcell_group_x = if (!is.na(group_col)) normalize_group(.data[[group_col]]) else NA_character_,
      pseudo_circle_id_x = if (!is.na(circle_col)) as.character(.data[[circle_col]]) else NA_character_,
      HNas_code_x = if (!is.na(hnas_col)) as.character(.data[[hnas_col]]) else pseudo_circle_id_x,
      HNas_code_clean_x = clean_hnas(HNas_code_x),
      xcell_celltype = as.character(.data[[celltype_col]]),
      xcell_score = as.numeric(.data[[score_col]])
    ) %>%
    dplyr::filter(
      !is.na(pseudo_sample_id),
      pseudo_sample_id != "",
      is.finite(xcell_score),
      !xcell_celltype %in% c("ImmuneScore", "StromaScore", "MicroenvironmentScore")
    )
  
  ##########################################################
  ## 4. 合并 sample_info，优先用 sample_info 的 section/group/HNas
  ##########################################################
  
  xraw <- xraw0 %>%
    dplyr::left_join(
      sample_meta,
      by = "pseudo_sample_id"
    ) %>%
    dplyr::mutate(
      module_label_original = module_label_original,
      module_label = module_label,
      module_key = norm_key(module_label),
      module_source_priority = module_source_priority(module_label_original),
      
      section = dplyr::case_when(
        !is.na(section_meta) & section_meta != "" ~ section_meta,
        !is.na(section_x) & section_x != "" ~ section_x,
        TRUE ~ NA_character_
      ),
      
      xcell_group = dplyr::case_when(
        !is.na(xcell_group_meta) & xcell_group_meta != "" ~ xcell_group_meta,
        !is.na(xcell_group_x) & xcell_group_x != "" ~ xcell_group_x,
        TRUE ~ NA_character_
      ),
      
      HNas_code_clean = dplyr::case_when(
        !is.na(HNas_code_clean_meta) & HNas_code_clean_meta != "" ~ HNas_code_clean_meta,
        !is.na(HNas_code_clean_x) & HNas_code_clean_x != "" ~ HNas_code_clean_x,
        TRUE ~ NA_character_
      ),
      
      n_spots_pseudo = ifelse(
        is.finite(n_spots_pseudo_meta),
        n_spots_pseudo_meta,
        1
      ),
      
      source_xcell_file = xcell_file
    ) %>%
    dplyr::left_join(
      celltype_to_category,
      by = "xcell_celltype"
    ) %>%
    dplyr::filter(
      cell_category %in% INFILTRATION_TYPES,
      section %in% c("LR_combined", "L", "R"),
      xcell_group %in% GROUP_ORDER,
      !is.na(HNas_code_clean),
      HNas_code_clean != ""
    ) %>%
    dplyr::select(
      module_label_original,
      module_label,
      module_key,
      module_source_priority,
      pseudo_sample_id,
      section,
      xcell_group,
      HNas_code_clean,
      xcell_celltype,
      xcell_score,
      cell_category,
      n_spots_pseudo,
      source_xcell_file,
      source_sample_info_file
    )
  
  xraw
}

calc_wilcox_infil <- function(df) {
  df %>%
    dplyr::group_by(module_label) %>%
    dplyr::summarise(
      n_total = dplyr::n(),
      n_low = sum(infiltration_group == "Low", na.rm = TRUE),
      n_high = sum(infiltration_group == "High", na.rm = TRUE),
      mean_low = mean(mean_module_score[infiltration_group == "Low"], na.rm = TRUE),
      mean_high = mean(mean_module_score[infiltration_group == "High"], na.rm = TRUE),
      p_value = tryCatch(
        wilcox.test(mean_module_score ~ infiltration_group)$p.value,
        error = function(e) NA_real_
      ),
      y_pos = {
        rg <- range(mean_module_score, na.rm = TRUE)
        if (!all(is.finite(rg)) || rg[1] == rg[2]) {
          max(mean_module_score, na.rm = TRUE) + 0.1
        } else {
          max(mean_module_score, na.rm = TRUE) + 0.08 * diff(rg)
        }
      },
      .groups = "drop"
    ) %>%
    dplyr::mutate(
      p_adj_BH = p.adjust(p_value, method = "BH"),
      p_label = make_p_label(p_value)
    )
}

plot_infil_box <- function(df, type_use) {
  
  stat_df <- calc_wilcox_infil(df)
  stat_df$infiltration_type <- type_use
  
  p <- ggplot(
    df,
    aes(
      x = infiltration_group,
      y = mean_module_score,
      fill = infiltration_group
    )
  ) +
    geom_boxplot(
      outlier.shape = NA,
      width = 0.60,
      linewidth = 0.30
    ) +
    geom_jitter(
      width = 0.16,
      size = 1.15,
      alpha = 0.72
    ) +
    geom_text(
      data = stat_df,
      aes(
        x = 1.5,
        y = y_pos,
        label = p_label
      ),
      inherit.aes = FALSE,
      size = 3.2,
      color = "black"
    ) +
    facet_wrap(
      ~ module_label,
      scales = "free_y",
      ncol = 3
    ) +
    scale_fill_manual(
      values = c(
        "Low" = "#4C78A8",
        "High" = "#E45756"
      )
    ) +
    theme_box(base_size = 11) +
    labs(
      title = paste0("Module score comparison between ", type_use, " high and low groups"),
      subtitle = "Grouping method: global circle-level 64 xCell category score; cutoff = median across circles",
      x = paste0(type_use, " group"),
      y = "Mean module score"
    )
  
  list(plot = p, stats = stat_df)
}

standardize_clin_group <- function(x, var) {
  y0 <- trimws(as.character(x))
  yl <- tolower(y0)
  
  if (var == "T_stage") {
    tval <- stringr::str_extract(toupper(y0), "T[1-4]")
    return(
      dplyr::case_when(
        tval %in% c("T1", "T2") ~ "T1-2",
        tval %in% c("T3", "T4") ~ "T3-4",
        TRUE ~ NA_character_
      )
    )
  }
  
  if (var == "N_stage") {
    nval <- stringr::str_extract(toupper(y0), "N[0-3]")
    return(
      dplyr::case_when(
        grepl("无|否|阴性|negative|no|none|0", yl) ~ "N0",
        grepl("有|是|阳性|positive|yes|1", yl) ~ "N+",
        nval == "N0" ~ "N0",
        nval %in% c("N1", "N2", "N3") ~ "N+",
        TRUE ~ NA_character_
      )
    )
  }
  
  if (var == "Recurrence") {
    return(
      dplyr::case_when(
        grepl("无复发|未复发|无|否|no|none|0", yl) ~ "No",
        grepl("复发|有|是|yes|1", yl) ~ "Yes",
        TRUE ~ NA_character_
      )
    )
  }
  
  if (var == "Differentiation") {
    return(
      dplyr::case_when(
        grepl("未分化", y0) ~ "非角化未分化型",
        grepl("分化", y0) ~ "非角化分化型",
        TRUE ~ y0
      )
    )
  }
  
  y0
}

group_order_by_var <- function(var) {
  if (var == "T_stage") return(c("T1-2", "T3-4"))
  if (var == "N_stage") return(c("N0", "N+"))
  if (var == "Recurrence") return(c("No", "Yes"))
  if (var == "Differentiation") return(c("非角化未分化型", "非角化分化型"))
  NULL
}

read_one_exact_clinical <- function(file, var) {
  
  message("读取临床文件：", var, " | ", file)
  
  raw <- readxl::read_excel(
    path = file,
    col_names = FALSE
  )
  
  raw <- as.data.frame(raw)
  
  if (ncol(raw) < 3) {
    stop("临床文件列数不足 3 列：", file)
  }
  
  df <- raw[, 1:3]
  colnames(df) <- c("region_id", "D07_sample_id", "clinical_value_raw")
  
  out <- df %>%
    dplyr::mutate(
      clinical_var = var,
      region_id = trimws(as.character(region_id)),
      D07_sample_id = trimws(as.character(D07_sample_id)),
      clinical_value_raw = trimws(as.character(clinical_value_raw)),
      clinical_group = standardize_clin_group(clinical_value_raw, var),
      source_file = file
    ) %>%
    dplyr::filter(
      !is.na(region_id),
      region_id != "",
      !is.na(D07_sample_id),
      D07_sample_id != "",
      !is.na(clinical_group),
      clinical_group != ""
    )
  
  out
}

calc_clinical_stats <- function(dd, var_use, levs_expected) {
  dd %>%
    dplyr::group_by(module_label) %>%
    dplyr::summarise(
      n_total = dplyr::n(),
      group_1 = levs_expected[1],
      group_2 = levs_expected[2],
      n_group_1 = sum(clinical_group == levs_expected[1], na.rm = TRUE),
      n_group_2 = sum(clinical_group == levs_expected[2], na.rm = TRUE),
      mean_group_1 = mean(mean_module_score[clinical_group == levs_expected[1]], na.rm = TRUE),
      mean_group_2 = mean(mean_module_score[clinical_group == levs_expected[2]], na.rm = TRUE),
      p_value = tryCatch(
        wilcox.test(mean_module_score ~ clinical_group)$p.value,
        error = function(e) NA_real_
      ),
      y_pos = {
        rg <- range(mean_module_score, na.rm = TRUE)
        if (!all(is.finite(rg)) || rg[1] == rg[2]) {
          max(mean_module_score, na.rm = TRUE) + 0.1
        } else {
          max(mean_module_score, na.rm = TRUE) + 0.08 * diff(rg)
        }
      },
      .groups = "drop"
    ) %>%
    dplyr::mutate(
      clinical_var = var_use,
      p_adj_BH = p.adjust(p_value, method = "BH"),
      p_label = make_p_label(p_value)
    )
}
